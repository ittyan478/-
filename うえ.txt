    local ok, val = pcall(function()
        return game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue()
    end)

local function GetPlayers(excludeLocalPlayer)
    local list = Players:GetPlayers()
    if excludeLocalPlayer then
        local idx = table.find(list, localPlayer)
        if idx then table.remove(list, idx) end
    end
    table.sort(list, function(a, b) return a.Name:lower() < b.Name:lower() end)
    return list
end

local function FormatPlayer(value)
    local player = value
    if type(value) == "string" then player = Players:FindFirstChild(value) end
    if not player then return tostring(value) end
    return string.format("%s (@%s)", player.DisplayName, player.Name)
end

local function OnTargetChanged(value)
    table.clear(Config.SelectedPlayers)
    table.clear(Config.PlayerList)
    if type(value) == "table" then
        for k, v in pairs(value) do
            local player = k
            if type(k) == "string" then
                local username = string.match(k, "%((.-)%)")
                if username then player = Players:FindFirstChild(username) end
            end
            if v and player and typeof(player) == "Instance" and player.Parent then
                Config.SelectedPlayers[player] = true
                table.insert(Config.PlayerList, player.Name)
            end
        end
    elseif value and typeof(value) == "Instance" and value.Parent then
        Config.SelectedPlayers[value] = true
        table.insert(Config.PlayerList, value.Name)
    elseif type(value) == "string" then
        local username = string.match(value, "%((.-)%)")
        local player = username and Players:FindFirstChild(username)
        if player then
            Config.SelectedPlayers[player] = true
            table.insert(Config.PlayerList, player.Name)
        end
    end
end

    local function AK_start()
        if AK.enabled then return end
        AK.enabled = true; AK.sitCount = 0; AK.antiActive = false; AK.isProcessing = false
        AK.isDead = false; AK.isRestarting = false; AK.isRagdolledTrue = false; AK.hasPartOwner = false
        if AK.respawnConnection then AK.respawnConnection:Disconnect(); AK.respawnConnection = nil end
        AK.respawnConnection = localPlayer.CharacterAdded:Connect(AK_onRespawn)
        AK.heartbeatConnection = RunService.Heartbeat:Connect(function()
            if not AK.enabled then
                if AK.heartbeatConnection then AK.heartbeatConnection:Disconnect(); AK.heartbeatConnection = nil end
                return
            end
            local hum = AK_getHumanoid()
            if hum and hum.Health <= 0 then AK.isDead = true; AK_stopAll(); return end
            if AK.isDead or AK.isRagdolledTrue then return end
            local head = AK_getHead()
            if head then
                local partOwner = head:FindFirstChild("PartOwner")
                if partOwner then
                    if not AK.hasPartOwner then
                        AK.hasPartOwner = true
                        AK_stopAll(); AK_destroyAll()
                        local hum2 = AK_getHumanoid()
                        if hum2 then hum2.Sit = false end
                    end
                else
                    if AK.hasPartOwner then AK.hasPartOwner = false; AK_restartFromSpawn() end
                end
            end
            if AK.hasPartOwner then return end
            local blobCount = AK_countBlobman()
            if blobCount == 0 then
                if AK.isRestarting then return end
                AK_stopAll()
                local hum2 = AK_getHumanoid()
                if hum2 then hum2.Sit = false end
                task.wait(0.05)
                if AK.enabled and not AK.isDead and not AK.isRagdolledTrue and not AK.hasPartOwner then AK_startBlobmanProcess() end
            elseif blobCount >= 2 then
                if AK.isRestarting then return end
                AK_stopAll(); AK_destroyAll()
                local hum2 = AK_getHumanoid()
                if hum2 then hum2.Sit = false end
                task.wait(0.05)
                if AK.enabled and not AK.isDead and not AK.isRagdolledTrue and not AK.hasPartOwner then AK_startBlobmanProcess() end
            end
        end)
        AK_startBlobmanProcess()
    end

    end

    -- Anti Kill Desync
    local DesyncSupported =
        type(sethiddenproperty) == "function" and type(setfflag) == "function" and type(getfflag) == "function"
    local function D_GetChar()
        local char = localPlayer.Character
        if not char or not char.Parent then return end
        local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("realHrp")
        local hum = char:FindFirstChildOfClass("Humanoid")

    local function D_SetRate(rate)
        for _, v in ipairs({ tostring(rate), rate }) do
            pcall(setfflag, "S2PhysicsSenderRate", v)
            local ok, res = pcall(getfflag, "S2PhysicsSenderRate")
            if ok and tonumber(res) == rate then return true end
        end
        return false
    end
    local function D_ResetState()
        DesyncSys.Epoch += 1
        if DesyncSys.Root and DesyncSys.Root.Parent then pcall(sethiddenproperty, DesyncSys.Root, "PhysicsRepRootPart", nil) end
        DesyncSys.Root = nil; DesyncSys.Busy = false; DesyncSys.NextToggle = 0
    end
    local D_SetDesync
    local function D_ForceDisable() if DesyncSys.Enabled then D_SetDesync(false) end end
    local function D_UpdateDesync()
        if not DesyncSys.Running or not DesyncSys.Enabled then return end
        local now = os.clock()
        if DesyncSys.RateGuardAt <= now then
            DesyncSys.RateGuardAt = now + 1
            local ok, res = pcall(getfflag, "S2PhysicsSenderRate")
            if (not ok or tonumber(res) ~= 1000) and not D_SetRate(1000) then return D_ForceDisable() end
        end
        local char, root, hum = D_GetChar()
        if not (char and root and hum and hum.Health > 0) then
            if DesyncSys.Root then D_ResetState() end
            return
        end
        if root ~= DesyncSys.Root then D_ResetState(); DesyncSys.Root = root; DesyncSys.NextToggle = now + 0.05 end
        if DesyncSys.Busy then return end
        if now < DesyncSys.NextToggle then
            if not pcall(sethiddenproperty, DesyncSys.Root, "PhysicsRepRootPart", DesyncSys.Root) then return D_ForceDisable() end
            return
        end
        DesyncSys.Busy = true
        local snapRoot = DesyncSys.Root
        local snapEp = DesyncSys.Epoch
        if not pcall(sethiddenproperty, snapRoot, "PhysicsRepRootPart", nil) then DesyncSys.Busy = false; return D_ForceDisable() end
        task.wait(0.04)
        if snapEp ~= DesyncSys.Epoch or not DesyncSys.Running or not DesyncSys.Enabled or snapRoot ~= DesyncSys.Root then
            DesyncSys.Busy = false
            return
        end
        if not snapRoot.Parent or char ~= localPlayer.Character or hum.Health <= 0 then
            DesyncSys.Busy = false
            return D_ResetState()
        end
        if not pcall(sethiddenproperty, snapRoot, "PhysicsRepRootPart", snapRoot) then DesyncSys.Busy = false; return D_ForceDisable() end
        DesyncSys.Busy = false
        DesyncSys.NextToggle = os.clock() + 0.05
    end
    D_SetDesync = function(state)
        local want = state == true and DesyncSys.Running and DesyncSupported
        if DesyncSys.Enabled == want then return end
        if DesyncSys.HeartbeatConn then DesyncSys.HeartbeatConn:Disconnect(); DesyncSys.HeartbeatConn = nil end
        DesyncSys.Enabled = want
        if want then
            local ok, res = pcall(getfflag, "S2PhysicsSenderRate")
            if ok and tonumber(res) then DesyncSys.OriginalRate = tonumber(res) end
            if DesyncSys.OriginalRate == nil then DesyncSys.Enabled = false; return D_ForceDisable() end
            DesyncSys.RateGuardAt = os.clock() + 1
            if not D_SetRate(1000) then DesyncSys.Enabled = false; return D_ForceDisable() end
            DesyncSys.HeartbeatConn = RunService.Heartbeat:Connect(D_UpdateDesync)
        else
            D_ResetState()
            if DesyncSys.OriginalRate ~= nil then D_SetRate(DesyncSys.OriginalRate) end
        end
    end
    table.insert(Defence._conns, localPlayer.CharacterAdded:Connect(function()
        D_ResetState()
        if DesyncSys.Enabled then DesyncSys.RateGuardAt = os.clock() + 1 end
    end))

    end

    local function AKS_ApplyStompFrame()
        if not aks_stompActive then return end
        local char = localPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        for _, p in ipairs(AKS_GetAllBaseParts(char)) do
            p.Anchored = true
            p.AssemblyLinearVelocity  = Vector3.zero
            p.AssemblyAngularVelocity = Vector3.zero
            pcall(function()
                p.Velocity    = Vector3.zero
                p.RotVelocity = Vector3.zero
            end)
        end
        if aks_stompFixedCFrame then
            for p, offset in pairs(aks_stompTorsoOffsets) do
                if p and p.Parent and offset then
                    p.CFrame = aks_stompFixedCFrame * offset
                    p.AssemblyLinearVelocity  = Vector3.zero
                    p.AssemblyAngularVelocity = Vector3.zero
                    pcall(function()
                        p.Velocity    = Vector3.zero
                        p.RotVelocity = Vector3.zero
                    end)
                end
            end
            for p, savedCF in pairs(aks_stompLimbCFrames) do
                if p and p.Parent and savedCF then
                    p.CFrame = savedCF
                    p.AssemblyLinearVelocity  = Vector3.zero
                    p.AssemblyAngularVelocity = Vector3.zero
                    pcall(function()
                        p.Velocity    = Vector3.zero
                        p.RotVelocity = Vector3.zero
                    end)
                end
            end
        end
        for p, reinf in pairs(aks_stompReinforce) do
            if p and p.Parent then
                if reinf.BP and reinf.BP.Parent then
                    reinf.BP.Position = p.Position
                end
                if reinf.BG and reinf.BG.Parent then
                    reinf.BG.CFrame = p.CFrame
                end
            end
        end
        local rootJoint = char:FindFirstChild("RootJoint", true)
        if rootJoint and aks_stompSavedJoints[rootJoint] then
            local saved = aks_stompSavedJoints[rootJoint]
            pcall(function()
                rootJoint.C0 = saved.C0
                rootJoint.C1 = saved.C1
            end)
        end
        hum.PlatformStand = true
        hum.WalkSpeed     = 0
        hum.JumpPower     = 0
        hum.JumpHeight    = 0
        hum.AutoRotate    = false
        if hum.Sit then hum.Sit = false end
    end
    local AKS_EndStomp
    AKS_EndStomp = function()
        if not aks_stompActive then return end
        aks_stompActive      = false
        aks_stompFixedCFrame = nil
        if aks_stompEndTask then
            pcall(function() task.cancel(aks_stompEndTask) end)
            aks_stompEndTask = nil
        end
        if aks_stompConnectionStep then aks_stompConnectionStep:Disconnect(); aks_stompConnectionStep = nil end
        if aks_stompConnectionBeat then aks_stompConnectionBeat:Disconnect(); aks_stompConnectionBeat = nil end
        if aks_stompConnectionRender then aks_stompConnectionRender:Disconnect(); aks_stompConnectionRender = nil end
        if aks_stompConnectionPre then aks_stompConnectionPre:Disconnect(); aks_stompConnectionPre = nil end
        for _, reinf in pairs(aks_stompReinforce) do
            if reinf.BP then pcall(function() reinf.BP:Destroy() end) end
            if reinf.BG then pcall(function() reinf.BG:Destroy() end) end
        end
        aks_stompReinforce = {}
        -- ★ バーストも同時停止（Stompより長く続くと無限ループの原因になる）
        AKS_StopAntiBlobmanKillBurst()
        AKS_StopRagdollBurst()
        local char = localPlayer.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            -- ★ 保存値がスタック状態(0等)でも安全な値へ強制復帰
            local ws = aks_stompSaved.WalkSpeed
            if not ws or ws <= 0 then ws = 16 end
            hum.WalkSpeed = ws
            local jp = aks_stompSaved.JumpPower
            if not jp or jp <= 0 then jp = 50 end
            hum.JumpPower = jp
            local jh = aks_stompSaved.JumpHeight
            if not jh or jh <= 0 then jh = 7.2 end
            hum.JumpHeight = jh
            hum.AutoRotate    = true
            hum.PlatformStand = false
            hum.Sit           = false
            -- ★ ラグドール状態から明示的に抜ける
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.Running) end)
        end
        if char then
            for p, saved in pairs(aks_stompSavedParts) do
                if p and p.Parent then
                    p.Anchored = saved.Anchored
                    p.AssemblyLinearVelocity  = Vector3.zero
                    p.AssemblyAngularVelocity = Vector3.zero
                end
            end
            local rj = char:FindFirstChild("RootJoint", true)
            if rj and aks_stompSavedJoints[rj] then
                local s = aks_stompSavedJoints[rj]
                pcall(function() rj.C0 = s.C0; rj.C1 = s.C1 end)
            end
        end
        aks_stompSavedParts    = {}
        aks_stompSavedJoints   = {}
        aks_stompTorsoOffsets  = {}
        aks_stompLimbCFrames   = {}
        aks_stompCooldownUntil = tick() + 0.15
    end
    local function AKS_StartStomp(character)
        if not character then return end
        if aks_stompActive then return end
        if tick() < aks_stompCooldownUntil then return end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root     = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not root then return end
        aks_stompActive  = true
        aks_stompEndTime = tick() + STOMP_DURATION
        AKS_TriggerAntiBlobmanKillBurst()
        AKS_TriggerRagdollBurst()
        aks_stompSaved = {
            WalkSpeed     = humanoid.WalkSpeed,
            JumpPower     = humanoid.JumpPower,
            JumpHeight    = humanoid.JumpHeight,
            AutoRotate    = humanoid.AutoRotate,
            PlatformStand = humanoid.PlatformStand,
        }
        aks_stompSavedParts   = {}
        aks_stompTorsoOffsets = {}
        aks_stompLimbCFrames  = {}
        local rootCF = root.CFrame
        for _, p in ipairs(AKS_GetAllBaseParts(character)) do
            aks_stompSavedParts[p] = { Anchored = p.Anchored }
            p.Anchored = true
            p.AssemblyLinearVelocity  = Vector3.zero
            p.AssemblyAngularVelocity = Vector3.zero
            if AKS_IsTorsoHeadPart(p) then
                aks_stompTorsoOffsets[p] = rootCF:ToObjectSpace(p.CFrame)
            else
                aks_stompLimbCFrames[p] = p.CFrame
            end
        end
        aks_stompSavedJoints = {}
        local rootJoint = character:FindFirstChild("RootJoint", true)
        if rootJoint then
            aks_stompSavedJoints[rootJoint] = { C0 = rootJoint.C0, C1 = rootJoint.C1 }
        end
        aks_stompFixedCFrame = root.CFrame
        aks_stompReinforce = {}
        local FORCE_MULT = 10e10 * STOMP_STRENGTH_MULT
        for _, p in ipairs(AKS_GetTorsoHeadParts(character)) do
            local bp = Instance.new("BodyPosition")
            bp.Name     = "AntiKillStompBP"
            bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            bp.P       = FORCE_MULT
            bp.D       = FORCE_MULT
            bp.Position = p.Position
            bp.Parent  = p
            local bg = Instance.new("BodyGyro")
            bg.Name      = "AntiKillStompBG"
            bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
            bg.P         = FORCE_MULT
            bg.D         = FORCE_MULT
            bg.CFrame    = p.CFrame
            bg.Parent    = p
            aks_stompReinforce[p] = { BP = bp, BG = bg }
        end
        humanoid.WalkSpeed     = 0
        humanoid.JumpPower     = 0
        humanoid.JumpHeight    = 0
        humanoid.AutoRotate    = false
        humanoid.PlatformStand = true
        if humanoid.Sit then humanoid.Sit = false end
        if aks_stompEndTask then
            pcall(function() task.cancel(aks_stompEndTask) end)
            aks_stompEndTask = nil
        end
        aks_stompEndTask = task.delay(STOMP_DURATION, AKS_EndStomp)
        if aks_stompConnectionStep then aks_stompConnectionStep:Disconnect(); aks_stompConnectionStep = nil end
        if aks_stompConnectionBeat then aks_stompConnectionBeat:Disconnect(); aks_stompConnectionBeat = nil end
        if aks_stompConnectionRender then aks_stompConnectionRender:Disconnect(); aks_stompConnectionRender = nil end
        if aks_stompConnectionPre then aks_stompConnectionPre:Disconnect(); aks_stompConnectionPre = nil end
        AKS_ApplyStompFrame()
        aks_stompConnectionPre = RunService.PreRender:Connect(function()
            if not aks_stompActive then return end
            AKS_ApplyStompFrame()
        end)
        aks_stompConnectionRender = RunService.RenderStepped:Connect(function()
            if not aks_stompActive then return end
            AKS_ApplyStompFrame()
        end)
        aks_stompConnectionBeat = RunService.Heartbeat:Connect(function()
            if not aks_stompActive then return end
            AKS_ApplyStompFrame()
        end)
        aks_stompConnectionStep = RunService.Stepped:Connect(function(_, _dt)
            if not aks_stompActive then return end
            AKS_ApplyStompFrame()
            if tick() >= aks_stompEndTime then AKS_EndStomp() end
        end)
    end
    local function AKS_FastDetect()
        if not AntiKillStompSys.active then return end
        local character = localPlayer.Character
        if not character then return end
        local grabbed, grabbedPart = AKS_IsGrabbed(character)
        local ragdolled = AKS_IsRagdolled(character)
        if grabbed or ragdolled then
            AKS_StartStomp(character)
        end
        if grabbed then
            if Remotes.Struggle then
                pcall(function() Remotes.Struggle:FireServer(localPlayer) end)
            end
            local humanoid = character:FindFirstChild("Humanoid")
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            if humanoid and rootPart and Remotes.RagdollRemote then
                pcall(function() Remotes.RagdollRemote:FireServer(rootPart, 0.00000000001) end)
            end
            if grabbedPart and grabbedPart:FindFirstChild("PartOwner") then
                pcall(function() grabbedPart.PartOwner.Value = "" end)
            end
            if humanoid and not aks_stompActive then
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
                humanoid.AutoRotate = true
                if humanoid.Sit then humanoid.Sit = false end
            end
        end
        AKS_StopGrabAnimation(character)
    end
    local function AKS_AntiKillLoop()
        while AntiKillStompSys.active and not Defence._unloaded do
            pcall(AKS_FastDetect)
            RunService.Stepped:Wait()
        end
    end
    local function StartAntiKillStomp()
        if AntiKillStompSys.active then return end
        AntiKillStompSys.active = true
        AntiKillStompSys.task = task.spawn(AKS_AntiKillLoop)
        AntiKillStompSys.fastConns = {
            RunService.Heartbeat:Connect(function()
                if AntiKillStompSys.active then pcall(AKS_FastDetect) end
            end),
            RunService.RenderStepped:Connect(function()
                if AntiKillStompSys.active then pcall(AKS_FastDetect) end
            end),
        }
    end
    local function StopAntiKillStomp()
        AntiKillStompSys.active = false
        if AntiKillStompSys.task then
            pcall(function() task.cancel(AntiKillStompSys.task) end)
            AntiKillStompSys.task = nil
        end
        for _, c in ipairs(AntiKillStompSys.fastConns or {}) do
            pcall(function() c:Disconnect() end)
        end
        AntiKillStompSys.fastConns = {}
        if aks_stompActive then AKS_EndStomp() end
        AKS_StopAntiBlobmanKillBurst()
        AKS_StopRagdollBurst()
    end

    local function AntiInputCleanupToy()
        local targetToy = defenceCfg.AntiInputType or "FoodHamburger"
        local toyFolder = workspace:FindFirstChild(localPlayer.Name .. "SpawnedInToys")
        local toy = toyFolder and toyFolder:FindFirstChild(targetToy)

    local function AntiInputGrabRelease(toy)
        local character = localPlayer.Character
        if not (toy and character) then return end
        local head = character:FindFirstChild("Head")
        if not head then return end
        local holdPart = toy:FindFirstChild("HoldPart")
        if not holdPart then return end
        local holdRemote = holdPart:FindFirstChild("HoldItemRemoteFunction")
        local dropRemote = holdPart:FindFirstChild("DropItemRemoteFunction")
        if holdRemote and dropRemote then
            local dropPosition = head.CFrame * CFrame.new(0, 3, 0)
            task.spawn(function()
                pcall(function()
                    holdRemote:InvokeServer(toy, character)
                    local dropDelay = defenceCfg.InputLgDelay or 0
                    if dropDelay > 0 then task.wait(dropDelay) end
                    dropRemote:InvokeServer(toy, dropPosition, dropPosition)
                end)
            end)
        end
    end
    local function AntiInputInitLoop()
        if antiInputV2.hbConn then return end
        antiInputV2.hbConn = RunService.Heartbeat:Connect(function()
            if not defenceState.AntiInputLag or Defence._unloaded then return end
            local targetToy = defenceCfg.AntiInputType or "FoodHamburger"
            local toyFolder = workspace:FindFirstChild(localPlayer.Name .. "SpawnedInToys")
            if not toyFolder then return end
            local toy = toyFolder:FindFirstChild(targetToy)
            if not toy then
                AntiInputCleanupToy()
                local char = localPlayer.Character
                local spawnCFrame = (char and char.PrimaryPart) and char.PrimaryPart.CFrame
                    or CFrame.new(0, 5, 0)
                if Remotes.SpawnToy then
                    pcall(function()
                        Remotes.SpawnToy:InvokeServer(targetToy, spawnCFrame, Vector3.new(0, 90, 0))
                    end)
                end
                for _ = 1, 15 do
                    toy = toyFolder:FindFirstChild(targetToy)
                    if toy then break end
                    task.wait()
                end
            end
            if toy then
                local transparency = defenceCfg.AILTransparency
                if transparency then
                    for _, part in ipairs(toy:GetDescendants()) do
                        if part:IsA("BasePart") and part.Transparency ~= transparency then
                            part.Transparency = transparency
                        end
                    end
                end
                AntiInputGrabRelease(toy)
            end
        end)
    end
    local function AntiInputTerminateLoop()
        if antiInputV2.hbConn then
            antiInputV2.hbConn:Disconnect()
            antiInputV2.hbConn = nil
        end
        AntiInputCleanupToy()
    end
    local function SetAntiInputLag(on)
        defenceState.AntiInputLag = on
        if on then
            AntiInputInitLoop()
            if not antiInputV2.respawnConn then
                antiInputV2.respawnConn = localPlayer.CharacterAdded:Connect(function()
                    if not defenceState.AntiInputLag then return end
                    task.wait(0.5)
                    AntiInputTerminateLoop()
                    task.wait(0.1)
                    AntiInputInitLoop()
                end)
            end
        else
            AntiInputTerminateLoop()
            if antiInputV2.respawnConn then
                antiInputV2.respawnConn:Disconnect()
                antiInputV2.respawnConn = nil
            end
        end
    end

            f_hrp, f_h, f_pin = hrp.CFrame, head.CFrame, hrp.CFrame
            RunService:BindToRenderStep("VoidPin", Enum.RenderPriority.Camera.Value - 1, function()
                hrp.CFrame = f_hrp; head.CFrame = f_h

                f_hrp, f_h = hrp.CFrame, head.CFrame

            end)

            pcall(RunService.UnbindFromRenderStep, RunService, "VoidPin")

            head.CFrame, hrp.CFrame = f_h, f_hrp

            end

            Vector3.new(0,0,radius), Vector3.new(0,0,-radius),

            Vector3.new(radius,0,-radius), Vector3.new(-radius,0,radius),

-- Ultimate Packet Lag v3 (8MB対応 / Bandwidth-Safe / Server-Heavy)
do
    local PacketLag = {
        Enabled = false,
        PacketSizeMB = 0.35,
        TargetBandwidthKBps = 40,
        Prefix = "RSC | ",
        AdaptiveMode = true,
        PacketLossChance = 0.05,
        PingThreshold = 250,
        StruggleFrameTime = 1 / 45,
        MaxSendRate = 450,
        MaxChunkBytes = 800,
        LoopToken = 0,
        FrameTimeAvg = 1 / 60,
        PingAvg = 0,
        LastHB = tick(),
        LastStatusAt = 0,
        LastPingUpdate = 0,
        TotalSent = 0,
        TotalSkipped = 0,
        TotalBytes = 0,
        RecentSends = {},
        StatusLabel = nil,
        PayloadString = nil,
    }

    local function rebuildPayload()
        local totalBytes = math.floor(PacketLag.PacketSizeMB * 1_000_000)
        local prefixLen = #PacketLag.Prefix
        local dataBytes = math.max(0, totalBytes - prefixLen)
        PacketLag.PayloadString = PacketLag.Prefix .. string.rep("0", dataBytes)
    end
    rebuildPayload()

    local function getPacketRemote()
        local grab = ReplicatedStorage:FindFirstChild("GrabEvents")
        return grab and grab:FindFirstChild("ExtendGrabLine")
    end

    local function getRecentBytes()
        local now = tick()
        local total = 0
        local recent = PacketLag.RecentSends
        for i = #recent, 1, -1 do
            if now - recent[i].t > 1 then
                table.remove(recent, i)
            else
                total = total + recent[i].b
            end
        end
        return total
    end

    local function getRecentCount()
        local now = tick()
        local count = 0
        for i = #PacketLag.RecentSends, 1, -1 do
            if now - PacketLag.RecentSends[i].t <= 1 then
                count = count + 1
            end
        end
        return count
    end

    local function getChunkSize()
        local logicalBytes = math.floor(PacketLag.PacketSizeMB * 1_000_000)
        return math.min(logicalBytes, PacketLag.MaxChunkBytes)
    end

    local function canSend(chunkBytes)
        local recentBytes = getRecentBytes()
        if recentBytes + chunkBytes > PacketLag.TargetBandwidthKBps * 1000 then
            return false
        end
        if getRecentCount() >= PacketLag.MaxSendRate then
            return false
        end
        return true
    end

    local function sendChunk(chunkBytes)
        local remote = getPacketRemote()
        if not remote then return false end
        local payload
        if chunkBytes >= #PacketLag.PayloadString then
            payload = PacketLag.PayloadString
        else
            payload = string.sub(PacketLag.PayloadString, 1, chunkBytes)
        end
        local ok = pcall(function() remote:FireServer(payload) end)
        if ok then
            table.insert(PacketLag.RecentSends, { t = tick(), b = chunkBytes })
            PacketLag.TotalSent = PacketLag.TotalSent + 1
            PacketLag.TotalBytes = PacketLag.TotalBytes + chunkBytes
        end
        return ok
    end

    local function updateFrameTime()
        local now = tick()
        local dt = now - PacketLag.LastHB
        PacketLag.LastHB = now
        if dt > 0 and dt < 0.5 then
            PacketLag.FrameTimeAvg = PacketLag.FrameTimeAvg * 0.85 + dt * 0.15
        end
    end

    local function updatePing()
        local now = tick()
        if now - PacketLag.LastPingUpdate < 0.5 then return end
        PacketLag.LastPingUpdate = now
        pcall(function()
            local ping = GetPacketPingMs()
            PacketLag.PingAvg = PacketLag.PingAvg * 0.7 + ping * 0.3
        end)
    end

    local function isStruggling()
        if PacketLag.FrameTimeAvg > PacketLag.StruggleFrameTime then return true end
        if PacketLag.PingAvg > PacketLag.PingThreshold then return true end
        return false
    end

    local function updateStatusText()
        if not PacketLag.StatusLabel then return end
        pcall(function()
            PacketLag.StatusLabel:SetText(string.format(
                "fps≈%.0f | ping≈%.0f | rate=%d/s | bw=%.0fKB/s | size=%.2fMB",
                1 / math.max(PacketLag.FrameTimeAvg, 0.001),
                PacketLag.PingAvg,
                getRecentCount(),
                getRecentBytes() / 1000,
                PacketLag.PacketSizeMB
            ))
        end)
    end

    local function setPacketLagEnabled(v)
        v = v == true
        PacketLag.Enabled = v
        PacketLag.LoopToken = PacketLag.LoopToken + 1
        if not v then return end
        local token = PacketLag.LoopToken
        local nextSendTime = tick()

        task.spawn(function()
            while PacketLag.Enabled and token == PacketLag.LoopToken do
                RunService.Heartbeat:Wait()
                local now = tick()
                updateFrameTime()
                updatePing()
                if now - PacketLag.LastStatusAt >= 0.25 then
                    PacketLag.LastStatusAt = now
                    updateStatusText()
                end
                if now >= nextSendTime then
                    local chunk = getChunkSize()
                    local skipProb = PacketLag.PacketLossChance
                    if PacketLag.AdaptiveMode and isStruggling() then
                        skipProb = 0.7
                    end
                    if math.random() < skipProb then
                        PacketLag.TotalSkipped = PacketLag.TotalSkipped + 1
                    elseif canSend(chunk) then
                        sendChunk(chunk)
                    else
                        PacketLag.TotalSkipped = PacketLag.TotalSkipped + 1
                    end
                    local interval = chunk / (PacketLag.TargetBandwidthKBps * 1000)
                    nextSendTime = now + interval
                end
            end
        end)
    end

    local PacketLagBox = Tabs.Server:AddRightGroupbox("Packet Lag", "wifi")
    PacketLagBox:AddToggle("PacketLagEnabled", {
        Text = "Enable Packet Lag", Default = false,
        Tooltip = "帯域制限50KB/sを超えずにサーバー負荷を最大化（8MBまで対応）",
        Callback = function(v) setPacketLagEnabled(v) end,
    })
    PacketLagBox:AddSlider("PacketSizeMB", {
        Text = "Packet Size", Default = 0.35, Min = 0.1, Max = 8, Rounding = 2,
        Suffix = " MB",
        Tooltip = "論理サイズ。実際の送信は帯域制限内で自動分割されます",
        Callback = function(v) PacketLag.PacketSizeMB = v; rebuildPayload() end,
    })
    PacketLagBox:AddSlider("PacketTargetBW", {
        Text = "Target Bandwidth", Default = 40, Min = 5, Max = 45, Rounding = 0,
        Suffix = " KB/s",
        Tooltip = "50KB/sがRobloxの上限。安全のため45以下推奨",
        Callback = function(v) PacketLag.TargetBandwidthKBps = v end,
    })
    PacketLagBox:AddSlider("PacketLossChance", {
        Text = "Packet Loss Chance", Default = 0.05, Min = 0, Max = 1, Rounding = 2,
        Callback = function(v) PacketLag.PacketLossChance = v end,
    })
    PacketLagBox:AddToggle("PacketAdaptiveMode", {
        Text = "Adaptive Mode", Default = true,

        Callback = function(v) PacketLag.AdaptiveMode = v end,
    })
    PacketLagBox:AddSlider("PacketMaxRate", {
        Text = "Max Send Rate", Default = 450, Min = 10, Max = 450, Rounding = 0,
        Suffix = " /s",
        Tooltip = "500/sがRobloxの上限。450以下推奨",
        Callback = function(v) PacketLag.MaxSendRate = v end,
    })
    PacketLagBox:AddDivider()
    PacketLagBox:AddButton({
        Text = "Send One Chunk", DoubleClick = false,
        Callback = function()
            local chunk = getChunkSize()
            if canSend(chunk) then sendChunk(chunk) end

    })
    PacketLagBox:AddButton({
        Text = "Reset Stats", DoubleClick = false,

            PacketLag.TotalSent = 0; PacketLag.TotalSkipped = 0

        end,

    do
        local s = PacketLagBox:AddLabel("fps≈60 | ping≈0 | rate=0/s | bw=0KB/s | size=0.35MB")

    end
    PacketLagBox:AddLabel("Packet Lag Key"):AddKeyPicker("PacketLagKeybind", {

                    task.spawn(function()

                        local camera = service.Workspace.CurrentCamera
                        local rayOrigin = camera.CFrame.Position
                        local rayDirection = camera.CFrame.LookVector * 1000
                        local rp = RaycastParams.new()
                        rp.FilterType = Enum.RaycastFilterType.Exclude
                        rp.FilterDescendantsInstances = { localPlayer.Character }
                        local rayResult = service.Workspace:Raycast(rayOrigin, rayDirection, rp)
                        if not rayResult then return end

                        local ignoredNames = { Map = true, Plots = true, Waypoints = true, Slots = true }

                            if ignoredNames[current.Name] then return end
                            current = current.Parent

                                    end)

                                    for i = 1, 18 do

                                        local targetPos = torso.Position

                                        if vel.Magnitude >= 2 then

                                        end
                                    until stop
                                end)

                                        SetNetworkOwner:FireServer(targetPart, targetPart.CFrame)
                                        task.wait(0.02)
                                    end
