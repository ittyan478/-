        if not char then return end
        for _, p in pairs(char:GetDescendants()) do
            if p:IsA("BasePart") then 
                p.Anchored = state 
                if not state then p.Velocity = Vector3.new(0,0,0) end
            end
        end
    end

    local function recover(hum, root)
        if not (hum and root) then return end
        pcall(function() root.AssemblyLinearVelocity = Vector3.new(0,0,0) end)
        pcall(function() root.AssemblyAngularVelocity = Vector3.new(0,0,0) end)
        pcall(function() hum.PlatformStand = false end)
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.Running) end)
    end

    local function setupAntiTrip(char)
        for _, c in pairs(tripConns) do c:Disconnect() end
        tripConns = {}
        
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not (hum and root) then return end

        local states = {
            Enum.HumanoidStateType.FallingDown, 
            Enum.HumanoidStateType.Ragdoll, 
            Enum.HumanoidStateType.PlatformStanding
        }
        for _, st in ipairs(states) do
            pcall(function() hum:SetStateEnabled(st, false) end)
        end

                recover(hum, root)
            end

                    local hrp = LP.Character:FindFirstChild("HumanoidRootPart")

do
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local Workspace = workspace

    local LocalPlayer = Players.LocalPlayer or Players.PlayerAdded:Wait()
    local playerToysFolderName = LocalPlayer.Name .. "SpawnedInToys"
    local targetObjectName = "FoodHamburger"

    local isAntiNetworkActive = false
    local teleportHeight = 15000000

    local function destroyTargetToy()
        local toyFolder = Workspace:FindFirstChild(playerToysFolderName)
        local targetToy = toyFolder and toyFolder:FindFirstChild(targetObjectName)
        if targetToy then
            local menuToys = ReplicatedStorage:FindFirstChild("MenuToys") or ReplicatedStorage.MenuToys
            menuToys.DestroyToy:FireServer(targetToy)
        end
    end

    RunService.Stepped:Connect(function()
        if isAntiNetworkActive then
            local toyFolder = Workspace:FindFirstChild(playerToysFolderName)
            local targetToy = toyFolder and toyFolder:FindFirstChild(targetObjectName)
            if targetToy then
                for _, part in pairs(targetToy:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                        part.CanTouch = false
                    end
                end
            end
        end
    end)

    local function claimOwnership(toyObject)
        local holdPart = toyObject:FindFirstChild("HoldPart")
        if not holdPart then return end
        
        local holdRemote = holdPart:FindFirstChild("HoldItemRemoteFunction")
        local dropRemote = holdPart:FindFirstChild("DropItemRemoteFunction")
        
        if holdRemote and dropRemote then
            local targetCFrame = CFrame.new(Vector3.new(0, teleportHeight, 0))
            task.spawn(function()
                holdRemote:InvokeServer(toyObject, LocalPlayer.Character)
                dropRemote:InvokeServer(toyObject, targetCFrame, targetCFrame)
            end)
        end
    end

    SettingsTab:AddSection({ Name = "Anti Network" })

    SettingsTab:AddToggle({
        Name = "Anti Network(food)",
        Default = false,
        Callback = function(inputValue)
            isAntiNetworkActive = inputValue
            
            if not inputValue then
                destroyTargetToy()
                return
            end

            task.spawn(function()
                while isAntiNetworkActive do
                    local toyFolder = Workspace:FindFirstChild(playerToysFolderName)
                    local targetToy = toyFolder and toyFolder:FindFirstChild(targetObjectName)
                    
                    if not targetToy then
                        local character = LocalPlayer.Character
                        local spawnCFrame = (character and character.PrimaryPart) and character.PrimaryPart.CFrame or CFrame.new(0, 5, 0)
                        
                        task.spawn(function()
                            local menuToys = ReplicatedStorage:FindFirstChild("MenuToys") or ReplicatedStorage.MenuToys
                            menuToys.SpawnToyRemoteFunction:InvokeServer(targetObjectName, spawnCFrame, Vector3.new(0, 90, 0))
                        end)
                        
                        local spawnedToy = nil
                        for _ = 1, 20 do
                            spawnedToy = toyFolder and toyFolder:FindFirstChild(targetObjectName)
                            if spawnedToy then break end
                            RunService.Heartbeat:Wait()
                        end
                        
                        if spawnedToy then 
                            claimOwnership(spawnedToy) 
                        end
                    else
                        claimOwnership(targetToy)
                    end
                    
                    RunService.Heartbeat:Wait()
                end
            end)

            task.spawn(function()
                while isAntiNetworkActive do
                    task.wait(12)
                    if isAntiNetworkActive then 
                        destroyTargetToy() 
                    end
                end
            end)
        end
    })

                            end)
                        end

    local EnvData = { 
        IsActive = false, 
        IsLoopRunning = false, 
        RagdollSpeed = 35, 
        JumpPower = 52 
    }

    local WantToJump = false

    local function findRemote()
        local rem = ReplicatedStorage:FindFirstChild("RagdollRemote")
        if rem then return rem end
        
        local folder = ReplicatedStorage:FindFirstChild("CharacterEvents")
        if folder then
            rem = folder:FindFirstChild("RagdollRemote")
            if rem then return rem end
        end
        
        return game:FindFirstChild("RagdollRemote", true)
    end

    local function setRagdollState(state)
        local character = LocalPlayer.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local remote = findRemote()
            if remote and remote:IsA("RemoteEvent") then
                remote:FireServer(hrp, state and 1 or 0)
            end
        end
    end

    local jumpConn = UserInputService.JumpRequest:Connect(function()
        if EnvData.IsActive then
            WantToJump = true
        end
    end)

    local function startRagdollLoop()
        if EnvData.IsLoopRunning then return end
        EnvData.IsLoopRunning = true
        
        while EnvData.IsActive do
            local character = LocalPlayer.Character
            local hrp = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChild("Humanoid")
            
            if hrp and humanoid then
                setRagdollState(true)
                
                local moveDir = humanoid.MoveDirection
                local velX = moveDir.X * EnvData.RagdollSpeed
                local velZ = moveDir.Z * EnvData.RagdollSpeed
                local velY = hrp.AssemblyLinearVelocity.Y
                
                if humanoid.Jump or WantToJump then
                    if math.abs(hrp.AssemblyLinearVelocity.Y) < 1.5 then
                        velY = EnvData.JumpPower
                    end
                    humanoid.Jump = false
                    WantToJump = false
                end

                if moveDir.Magnitude > 0 or velY == EnvData.JumpPower then
                    hrp.AssemblyLinearVelocity = Vector3.new(velX, velY, velZ)
                end
            end
            task.wait()
        end
        
        EnvData.IsLoopRunning = false
        WantToJump = false
        setRagdollState(false)
    end

    SettingsTab:AddToggle({
        Name = "Ragdoll (No Gucci)",
        Default = false,
        Callback = function(value)
            EnvData.IsActive = value
            if value then
                task.spawn(startRagdollLoop)
            end
        end
    })
end

local function ApplyCharacterSettings(character)
    local humanoid = character:WaitForChild("Humanoid", 5)
    if humanoid then
        -- トグルがONなら指定値、OFFならデフォルト値を適用
        humanoid.WalkSpeed = _G.SpeedEnabled and _G.TargetSpeed or 16
        humanoid.UseJumpPower = true
        humanoid.JumpPower = _G.JumpEnabled and _G.TargetJump or 50
        humanoid.JumpHeight = _G.JumpEnabled and _G.TargetJump or 50
        
        -- 強制書き換え検知イベント（トグル状態を常に反映）
        humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function() 
            humanoid.WalkSpeed = _G.SpeedEnabled and _G.TargetSpeed or 16 
        end)
        humanoid:GetPropertyChangedSignal("JumpPower"):Connect(function() 
            humanoid.JumpPower = _G.JumpEnabled and _G.TargetJump or 50 
        end)
        humanoid:GetPropertyChangedSignal("JumpHeight"):Connect(function() 
            humanoid.JumpHeight = _G.JumpEnabled and _G.TargetJump or 50 
        end)
        
        humanoid:GetPropertyChangedSignal("Parent"):Connect(function()
            if humanoid.Parent then 
                humanoid.WalkSpeed = _G.SpeedEnabled and _G.TargetSpeed or 16 
            end
        end)
    end
    
    if Workspace.CurrentCamera then
        Workspace.CurrentCamera.FieldOfView = _G.TargetFOV
    end
end

-- 毎フレームの超高速強制同期・物理制御ループ
if _G.MainExecutionLoop then _G.MainExecutionLoop:Disconnect() end
_G.MainExecutionLoop = RunService.RenderStepped:Connect(function()
    pcall(function()
        local char = LocalPlayer.Character
        if char and char:FindFirstChildOfClass("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
            local hum = char:FindFirstChildOfClass("Humanoid")
            local hrp = char.HumanoidRootPart
            
            -- 1. 移動速度管理（トグルがONの時のみ上書き固定）
            local currentTargetSpeed = _G.SpeedEnabled and _G.TargetSpeed or 16
            if hum.WalkSpeed ~= currentTargetSpeed then hum.WalkSpeed = currentTargetSpeed end
            
            -- 2. ジャンプ力管理（トグルがONの時のみ上書き固定）
            local currentTargetJump = _G.JumpEnabled and _G.TargetJump or 50
            if hum.JumpPower ~= currentTargetJump then hum.JumpPower = currentTargetJump end
            if hum.JumpHeight ~= currentTargetJump then hum.JumpHeight = currentTargetJump end
            
            -- 3. 画角管理
            if Workspace.CurrentCamera and Workspace.CurrentCamera.FieldOfView ~= _G.TargetFOV then
                Workspace.CurrentCamera.FieldOfView = _G.TargetFOV
            end
            
            -- 4. 【精度極限強化】新型高精度フライ処理
            if _G.FlyEnabled then
                -- 物理的な落下・慣性を完全に遮断して空中静止させる
                hrp.AssemblyLinearVelocity = Vector3.zero
                
                local flyDir = Vector3.zero
                local camCF = Workspace.CurrentCamera.CFrame
                
                -- キー入力判定
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then flyDir = flyDir + camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then flyDir = flyDir - camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then flyDir = flyDir + camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then flyDir = flyDir - camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then flyDir = flyDir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then flyDir = flyDir - Vector3.new(0, 1, 0) end
                
                -- 移動している時だけ滑らかに座標をずらす（離せばその場で完全ピタ止まり）
                if flyDir.Magnitude > 0 then
                    hrp.CFrame = CFrame.new(hrp.Position + (flyDir.Unit * (_G.FlySpeed / 50))) * camCF.Rotation
                else
                    -- 静止時はカメラの向きに合わせてキャラクターの回転だけを同期
                    hrp.CFrame = CFrame.new(hrp.Position) * camCF.Rotation
                end
            end
        end
    end)
end)

-- 三人称カメラの永続制御ループ
if _G.ThirdPersonLoop then _G.ThirdPersonLoop:Disconnect() end
_G.ThirdPersonLoop = RunService.RenderStepped:Connect(function()
    pcall(function()
        -- フライ機能ON時はカメラ競合を防ぐため一時スキップ
        if _G.ThirdPersonEnabled and not _G.FlyEnabled then
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            
            if hrp and hum and hum.Health > 0 then
                -- カメラタイプをスクリプト制御（Scriptable）に強制変更し、Roblox標準のキャラ透明化処理を完全に無効化
                if Workspace.CurrentCamera.CameraType ~= Enum.CameraType.Scriptable then
                    Workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable
                end

                -- キャラクターの描画を強制的に可視化（ローローカルでの透明化バグを上書き）
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") and part.LocalTransparencyModifier > 0 then
                        part.LocalTransparencyModifier = 0
                    end
                end

                -- マウス・画面ドラッグによる回転値のみをクリーンに取得
                local camRotation = Workspace.CurrentCamera.CFrame - Workspace.CurrentCamera.CFrame.Position
                local camLook = Workspace.CurrentCamera.CFrame.LookVector
                
                -- キャラクターの背後から「距離」と「上下左右のオフセット」を正確に計算した位置にカメラを配置
                local targetCamCFrame = CFrame.new(hrp.Position) 
                    * camRotation 
                    * CFrame.new(_G.CameraOffsetH, _G.CameraOffsetV, _G.CameraDistance)
                
                Workspace.CurrentCamera.CFrame = targetCamCFrame
                
                -- 移動時、または静止時もキャラの向きをカメラの正面（水平方向）に完全同期
                local flatLook = Vector3.new(camLook.X, 0, camLook.Z).Unit
                hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + flatLook)
            end
        else
            -- 三人称がOFFの時は、Roblox標準の自由カメラ（Custom）に瞬時に戻す
            if Workspace.CurrentCamera.CameraType == Enum.CameraType.Scriptable then
                Workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
            end
        end
    end)
end)

local function FireNetwork(part)
    if not part then return end
    pcall(function()
        local grabEvents = ReplicatedStorage:FindFirstChild("GrabEvents")
        if grabEvents and grabEvents:FindFirstChild("SetNetworkOwner") then
            grabEvents.SetNetworkOwner:FireServer(part, CFrame.new(part.Position))
        end
    end)
end

local function GetNearbyPlayer()
    local localChar = me.Character
    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then return nil end
    local localRoot = localChar.HumanoidRootPart
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player == me then continue end
        if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then continue end
        if whiteFriend and me:IsFriendsWith(player.UserId) then continue end
        
        local targetRoot = player.Character.HumanoidRootPart
        local distance = (localRoot.Position - targetRoot.Position).Magnitude
        
        if distance <= DETECT_RANGE then
            return player, targetRoot
        end
    end
    return nil
end

local function StartDetection(mode)
    currentMode = mode
    if loopConnection then loopConnection:Disconnect() end
    
    loopConnection = RunService.Heartbeat:Connect(function()
        if not isRunning then return end
        
        local player, targetRoot = GetNearbyPlayer()
        
        if player and not processingPlayers[player.Name] then
            processingPlayers[player.Name] = true
            
            task.spawn(function()
                local currentAngle = 0
                local isInfinite = (mode == "GrabUp")
                local iterations = isInfinite and 999999 or 50 

                for i = 1, iterations do
                    if not isRunning or currentMode ~= mode then break end
                    if not targetRoot or not targetRoot.Parent then break end
                    
                    FireNetwork(targetRoot)
                    
                    if mode == "GrabUp" then
                        targetRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 1.2, 0)
                    elseif mode == "Fling" then
                        if i == 10 then 
                            targetRoot.CFrame = CFrame.new(0, 1000000000000, 0) 
                            break 
                        end
                    elseif mode == "Orbit" then
                        currentAngle = currentAngle + math.rad(orbitSpeed)
                        local selfRoot = me.Character and me.Character:FindFirstChild("HumanoidRootPart")
                        if selfRoot then
                            local offset = Vector3.new(
                                math.cos(currentAngle) * orbitRadius,
                                2,
                                math.sin(currentAngle) * orbitRadius
                            )
                            targetRoot.CFrame = CFrame.new(selfRoot.Position + offset)
                        end
                    elseif mode == "Normal" then
                        targetRoot.CFrame = CFrame.new(targetRoot.Position)
                    end
                    
                    task.wait(NET_INTERVAL)
                end
                
                processingPlayers[player.Name] = nil
            end)
        end
    end)
end

-- ===== 機能①：Loop TP to Target（超高速・スムーズ） =====
local loopTPActive = false
local loopTPTask = nil

local function LoopTPFunction()
    local player = game:GetService("Players").LocalPlayer
    local RunService = game:GetService("RunService")
    
    while loopTPActive do
        local tChar = GetSelectedTargetCharacter()
        local myChar = player.Character
        if tChar and myChar then
            local tRoot = tChar:FindFirstChild("HumanoidRootPart")
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            if tRoot and myRoot then
                local targetCF = tRoot.CFrame * CFrame.new(3, 0, 0)
                myRoot.CFrame = myRoot.CFrame:Lerp(targetCF, 0.9) -- 超高速
                myRoot.Velocity = Vector3.zero
            end
        end
        RunService.Heartbeat:Wait()
    end
end

-- ===== 機能②：Spectate Target（相手の視点を完全に映す観戦モード） =====
local spectateActive = false
local spectateConnection = nil

local function ResetCamera()
    local player = game:GetService("Players").LocalPlayer
    local myChar = player.Character
    if myChar and myChar:FindFirstChild("HumanoidRootPart") then
        local hrp = myChar.HumanoidRootPart
        workspace.CurrentCamera.CFrame = CFrame.lookAt(
            hrp.Position + Vector3.new(0, 1, 5),
            hrp.Position
        )
    end
end

local function StartSpectate()
    local RunService = game:GetService("RunService")
    local camera = workspace.CurrentCamera

    spectateConnection = RunService.RenderStepped:Connect(function()
        local tChar = GetSelectedTargetCharacter()
        if not tChar then return end

        -- 相手の頭（なければHumanoidRootPart）
        local targetPart = tChar:FindFirstChild("Head") or tChar:FindFirstChild("HumanoidRootPart")
        if not targetPart then return end

        -- ★ 相手のHeadのCFrameをそのままカメラに反映（完全な一人称視点）
        -- ただし、少し後ろに下げてクリッピング防止（-lookDir * 0.3）
        local lookDir = targetPart.CFrame.LookVector
        local camPos = targetPart.Position - lookDir * 0.3  -- 頭の少し後ろ
        local targetCF = CFrame.lookAt(camPos, targetPart.Position + lookDir * 100)  -- 100スタッド先を見る

        -- ★ 滑らかに追従（Lerp 係数 0.35 → 速すぎず遅すぎず）
        camera.CFrame = camera.CFrame:Lerp(targetCF, 0.35)
    end)
end

-- 2.5. Auto Sit (CreatureBlobman) 限界強化版
-- ==========================================
do
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer or Players.PlayerAdded:Wait()

    local autoSitActive = false
    local autoSitLoop = nil

    -- 高速・安全に座らせるコア関数
    local function ForceSit(char, hum, seat)
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp or hum.SeatPart == seat then return true end
        
        -- 物理演算と座標を完全に座席に同期
        hrp.CFrame = seat.CFrame + Vector3.new(0, 1.2, 0)
        hrp.Velocity = Vector3.zero
        
        local success = pcall(function()
            seat:Sit(hum)
        end)
        return success and hum.SeatPart == seat
    end

    local function AutoSitLoopFunc()
        -- 自分のトイフォルダ名のキャッシュ（文字列結合のオーバーヘッド削減）
        local folderName = LocalPlayer.Name .. "SpawnedInToys"
        
        while autoSitActive do
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            
            if not char or not hum or hum.Health <= 0 then
                task.wait(0.5)
                continue
            end

            -- すでに何か（キル用のブロブマン等）に座っている場合はループを維持して待機
            if hum.SeatPart then
                RunService.Heartbeat:Wait()
                continue
            end

            local folder = workspace:FindFirstChild(folderName)
            local blob = folder and folder:FindFirstChild("CreatureBlobman")

            -- ブロブマンが存在しない場合、最速サーバーリクエスト
            if not blob then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    pcall(function()
                        -- 自分の少し上にスポーンさせ、即時検知・着席の成功率をアップ
                        local spawnPos = hrp.CFrame * CFrame.new(0, 2, -3)
                        ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", spawnPos, Vector3.zero)
                    end)
                    
                    -- WaitForChildのタイムアウトを短縮しラグを徹底排除
                    if not folder then folder = workspace:WaitForChild(folderName, 1.5) end
                    if folder then blob = folder:WaitForChild("CreatureBlobman", 1.5) end
                end
            end

            -- ブロブマン発見後の超高速着席シーケンス
            if blob then
                local seat = blob:FindFirstChildOfClass("VehicleSeat") or blob:FindFirstChildOfClass("Seat")
                if seat then
                    -- サーバー側の認識ズレを貫通するバースト試行（最大15回、フレーム同期）
                    for i = 1, 15 do
                        if not autoSitActive or hum.SeatPart or not seat.Parent then break end
                        
                        local sat = ForceSit(char, hum, seat)
                        if sat then break end
                        
                        RunService.Heartbeat:Wait() -- task.wait(0.3)からフレーム同期へ変更し超高速化
                    end
                end
            end
            
            -- 次のチェックまでのインターバルを最適化（ラグを防ぎつつ負荷を抑える絶妙な値）
            task.wait(0.2)
        end
    end

    local function StartAutoSit()
        if autoSitLoop then task.cancel(autoSitLoop) end
        autoSitLoop = task.spawn(AutoSitLoopFunc)
    end

    local function StopAutoSit()
        if autoSitLoop then 
            task.cancel(autoSitLoop)
            autoSitLoop = nil 
        end
    end

    -- TargetTabのトグルのすぐ下に追加
    TargetTab:AddToggle({
        Name = " Auto  sit blobman",
        Default = false,
        Callback = function(Value)
            autoSitActive = Value
            if Value then
                StartAutoSit()
                OrionLib:MakeNotification({Name = "auto sit", Content = "ON", Time = 1.5})
            else
                StopAutoSit()
                OrionLib:MakeNotification({Name = "auto sit", Content = "OFF", Time = 1})
            end
        end
    })
end

                    blob.BlobmanSeatAndOwnerScript.CreatureRelease:FireServer(blob.LeftDetector.LeftWeld)

-- 乗っているブロブマンを取得する関数（変更なし）
local function GetRiddenBlobman()
    local char = game:GetService("Players").LocalPlayer.Character
    if not char then return nil end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return nil end
    local seat = humanoid.SeatPart
    if not seat then return nil end
    local blob = seat.Parent
    if blob and blob.Name == "CreatureBlobman" and blob:FindFirstChild("BlobmanSeatAndOwnerScript") then

    end
    return nil
end

local loopGrabReleaseActive = false
local loopGrabReleaseTask = nil

local function LoopGrabReleaseFunction(targetName)
    local target = game:GetService("Players"):FindFirstChild(targetName)
    if not target then return end

    while loopGrabReleaseActive and target do
        local tChar = target.Character
        local myChar = game:GetService("Players").LocalPlayer.Character

        if tChar and myChar then
            local tRoot = tChar:FindFirstChild("HumanoidRootPart")
            local tHum = tChar:FindFirstChildOfClass("Humanoid")
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            local currentBlob = GetRiddenBlobman()

            if myRoot and tRoot and tHum and tHum.Health > 0 and currentBlob then

                -- ===== 1. 自分をターゲットの真横に超高速テレポート（ループで連続移動） =====
                myRoot.CFrame = tRoot.CFrame * CFrame.new(3, 0, 0)
                myRoot.Velocity = Vector3.zero

                -- ===== 2. ネットワークオーナー強奪（念のため） =====
                pcall(function()
                    local ReplicatedStorage = game:GetService("ReplicatedStorage")
                    local SetNetworkOwner = ReplicatedStorage:FindFirstChild("GrabEvents") and ReplicatedStorage.GrabEvents:FindFirstChild("SetNetworkOwner")
                    if SetNetworkOwner then
                        for i = 1, 5 do
                            SetNetworkOwner:FireServer(tRoot, myRoot.CFrame)
                        end
                    end
                end)

                -- ===== 3. グラブリリースを開始する直前に「ターゲットを自分の前に固定」＆「キル」 =====
                -- 固定：ターゲットを自分の真正面に強制移動＋速度リセット（動けなくする）
                tRoot.CFrame = myRoot.CFrame * CFrame.new(0, 0, -5) -- 自分の前5スタッド
                tRoot.Velocity = Vector3.zero

                -- キル：即座に死亡させる
                pcall(function()
                    if tHum then
                        tHum.Health = 0
                    end
                    if tChar then
                        tChar:BreakJoints()
                    end
                end)

                -- ===== 4. グラブリリースを１０回連続実行（開始直後に固定＆キル済み） =====
                for i = 1, 10 do
                    -- Grab
                    pcall(function()
                        currentBlob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                            currentBlob.LeftDetector,
                            tRoot,
                            currentBlob.LeftDetector.LeftWeld
                        )
                    end)

                    -- Release
                    pcall(function()
                        currentBlob.BlobmanSeatAndOwnerScript.CreatureRelease:FireServer(currentBlob.LeftDetector.LeftWeld)
                    end)

                    if i < 10 then
                        task.wait(0.05) -- 内部ループも高速化
                    end
                end

                -- 念のため、最後にもう一度位置をリセット（固定維持）
                if tRoot and myRoot then
                    tRoot.CFrame = myRoot.CFrame * CFrame.new(0, 0, -5)
                    tRoot.Velocity = Vector3.zero
                end
            end
        end

        -- メインループのウェイトを短くして超高速ループを実現
        task.wait(0.05)
    end
end

    local GE = game:GetService("ReplicatedStorage"):FindFirstChild("GrabEvents")
    local SetNetworkOwner = GE and GE:FindFirstChild("SetNetworkOwner")
    local CreateGrabLine = GE and GE:FindFirstChild("CreateGrabLine")
    local DestroyGrabLine = GE and GE:FindFirstChild("DestroyGrabLine")
    
    local lastAttackedChar = nil
    
    while loopNormalKillActive and target do
        local tChar = target.Character
        -- 新しくリスポーンした時だけ実行
        if tChar and tChar ~= lastAttackedChar then
            local tRoot = tChar:FindFirstChild("HumanoidRootPart")
            local tHum = tChar:FindFirstChildOfClass("Humanoid")
            
            local myChar = game:GetService("Players").LocalPlayer.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

            if myRoot and tRoot and tHum and tHum.Health > 0 then
                -- 1. ターゲットのすぐ後ろにテレポート
                myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                myRoot.Velocity = Vector3.zero
                
                -- 2. ネットワークオーナー強奪（掴む準備）
                pcall(function()
                    if SetNetworkOwner then
                        for i = 1, 3 do
                            SetNetworkOwner:FireServer(tRoot, myRoot.CFrame)
                        end
                    end
                end)
                
                -- 3. 普通にグラブで掴む
                pcall(function()
                    if CreateGrabLine then
                        CreateGrabLine:FireServer(tRoot, Vector3.zero, tRoot.Position, false)
                    end
                end)

                -- 4. 掴んだ状態で少し待機（0.4秒間相手の後ろをキープ）
                local attackStart = tick()
                while tick() - attackStart < 0.4 and tRoot.Parent and loopNormalKillActive do
                    myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                    myRoot.Velocity = Vector3.zero
                    game:GetService("RunService").Heartbeat:Wait()
                end
                
                -- 5. 掴んだままキルする
                pcall(function()
                    if tRoot.Parent then
                        -- 念のため再度ネットワーク強奪
                        if SetNetworkOwner then
                            SetNetworkOwner:FireServer(tRoot, myRoot.CFrame)
                        end
                        
                        -- キル実行
                        if tHum then
                            tHum:ChangeState(Enum.HumanoidStateType.Dead)
                            tHum.Health = 0
                        end
                        if tChar then
                            tChar:BreakJoints()
                        end
                        
                        -- 用が済んだらグラブを離す
                        if DestroyGrabLine then
                            DestroyGrabLine:FireServer(tRoot)
                        end
                    end
                end)
                
                -- キル実行済みとして記録
                lastAttackedChar = tChar
            end
        end
        task.wait(0.2) -- 監視ループ間隔
    end
end

-- ===== Ownership Kick（既存の selectedTarget を使用） =====
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local LocalPlayer = Players.LocalPlayer

local ownershipKickActive = false
local ownershipKickTask = nil
local fpsConnection = nil
local bodyPos, bodyGyro = nil, nil

local function cleanupBodies()
    pcall(function()
        if bodyPos then bodyPos:Destroy() bodyPos = nil end
        if bodyGyro then bodyGyro:Destroy() bodyGyro = nil end
    end)
end

local function OwnershipKickFunction(targetName)
    local target = Players:FindFirstChild(targetName)
    if not target then return end
    local GE = ReplicatedStorage:WaitForChild("GrabEvents")
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local savedPos = myRoot.CFrame
    local dragging = false
    local grabStartTime, checkStartTime = 0, 0
    local currentFPS = 60

    if fpsConnection then fpsConnection:Disconnect() end
    fpsConnection = RunService.RenderStepped:Connect(function(dt)
        currentFPS = 1 / dt
    end)

    local function createBodies(targetRoot, pos)
        cleanupBodies()
        for _, v in pairs(targetRoot:GetChildren()) do
            if v:IsA("BodyPosition") or v:IsA("BodyGyro") then v:Destroy() end
        end
        bodyPos = Instance.new("BodyPosition")
        bodyPos.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        bodyPos.D = 100
        bodyPos.Position = pos
        bodyPos.Parent = targetRoot

        bodyGyro = Instance.new("BodyGyro")
        bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        bodyGyro.D = 100
        bodyGyro.CFrame = CFrame.new(pos)
        bodyGyro.Parent = targetRoot
    end

    while ownershipKickActive do
        local currentTarget = Players:FindFirstChild(targetName)
        if not currentTarget then cleanupBodies() break end

        myChar = LocalPlayer.Character
        myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local tChar = currentTarget.Character
        local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
        local tHum = tChar and tChar:FindFirstChild("Humanoid")

        if tRoot and tHum and tHum.Health > 0 and myRoot then
            if not dragging then
                myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3) -- 右に3スタッド（必要に応じて X に変更）
                cleanupBodies()
                checkStartTime = 0
                pcall(function()
                    tHum.PlatformStand = true
                    tHum.Sit = true
                    GE.SetNetworkOwner:FireServer(tRoot, tRoot.CFrame)
                    GE.SetNetworkOwner:FireServer(tRoot, tRoot.CFrame)
                    GE.DestroyGrabLine:FireServer(tRoot)
                end)
                myRoot.AssemblyLinearVelocity = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
                if grabStartTime == 0 then grabStartTime = tick() end
                if tick() - grabStartTime > 0.35 then
                    dragging = true
                    grabStartTime = 0
                    checkStartTime = tick()
                    local lockPos = savedPos * CFrame.new(0, 17, 0)
                    createBodies(tRoot, lockPos.Position)
                end
            else
                myRoot.CFrame = savedPos
                local lockPos = savedPos * CFrame.new(0, 17, 0)
                myRoot.AssemblyLinearVelocity = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
                if bodyPos and bodyPos.Parent then
                    bodyPos.Position = lockPos.Position
                    if bodyGyro then bodyGyro.CFrame = lockPos end
                else
                    createBodies(tRoot, lockPos.Position)
                end
                tHum.PlatformStand = true
                pcall(function()
                    if currentFPS > 200 then
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.DestroyGrabLine:FireServer(tRoot)
                    elseif currentFPS >= 155 then
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.DestroyGrabLine:FireServer(tRoot)
                    else
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                        GE.DestroyGrabLine:FireServer(tRoot)
                    end
                end)
                if checkStartTime > 0 and tick() - checkStartTime > 0.30 then
                    if (tRoot.Position - lockPos.Position).Magnitude > 10 then
                        dragging = false
                        grabStartTime, checkStartTime = 0, 0
                        cleanupBodies()
                        myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                    else
                        checkStartTime = tick()
                    end
                end
            end
        else
            dragging = false
            grabStartTime, checkStartTime = 0, 0
            cleanupBodies()
        end
        RunService.Heartbeat:Wait()
    end

    if fpsConnection then fpsConnection:Disconnect() fpsConnection = nil end
    cleanupBodies()
    if myRoot then myRoot.CFrame = savedPos end
end

        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local hum = char:WaitForChild("Humanoid")

                warn("Stopped: left Blobman")
                break

            blobRoot = blob:FindFirstChild("HumanoidRootPart") or blob.PrimaryPart

            local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")

                        pcall(function()
                            tHum.PlatformStand = true
                            tHum.Sit = true
                            GE.SetNetworkOwner:FireServer(tRoot, blobRoot.CFrame)
                            GE.DestroyGrabLine:FireServer(tRoot)

            local hum = char and char:FindFirstChildOfClass("Humanoid")
            
            if hrp and hum and hum.Health > 0 then
                local cache = SetupPlayerCache(p)

                -- 枠線 (Highlight)

                            if cache.Billboard then cache.Billboard:Destroy() end
                            local bGui = Instance.new("BillboardGui")
                            bGui.Name = "Tactical_Tag"
                            bGui.Size = UDim2.new(0, 200, 0, 100)
                            bGui.AlwaysOnTop = true

                    end
                elseif cache.Billboard then
                    cache.Billboard.Enabled = false
                end

                -- ツール持ちの肥大化を防ぐため、HumanoidRootPart基準の正確な固定サイズ(4.5, 5.5, 4.5)に修正
                if _G.PlayerBoxEnabled then

                    end

                -- 1階層下まで確実に再帰検索（ラグらないように子要素のみに限定）
                for _, child in ipairs(target:GetChildren()) do
                    if child.Name == "CreatureBlobman" then checkAndApply(child) end
                end
            end
            checkAndApply(obj)
        end
    else
        -- オフの時は描画パーツを非表示にするだけ（ラグ防止）
        for _, cache in pairs(blobmanEspCache) do
            if cache.Highlight then cache.Highlight.Enabled = false end
            if cache.Billboard then cache.Billboard.Enabled = false end
        end
    end
end)

local function getAllPlayers()
    local players = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local isFriend = false
            pcall(function()
                isFriend = LocalPlayer:IsFriendsWith(plr.UserId)
            end)
            
            if not isFriend then
                table.insert(players, plr)
            end
        end
    end
    return players
end

local function spamOwnership(hrp)
    if not GrabEvents then return end
    local setOwner = GrabEvents:FindFirstChild("SetNetworkOwner")
    if setOwner and hrp then
        pcall(function() setOwner:FireServer(hrp, hrp.CFrame) end)
    end
end

local function teleportToPlayer(myHrp, targetHrp)
    if not myHrp or not targetHrp then return end
    pcall(function()
        myHrp.CFrame = targetHrp.CFrame * CFrame.new(0, 5, 5)
        myHrp.AssemblyLinearVelocity = Vector3.zero
    end)
end

local function destroyLineOnPlayer(hrp)
    if not GrabEvents then return end
    local createLine = GrabEvents:FindFirstChild("CreateGrabLine")
    local destroyLine = GrabEvents:FindFirstChild("DestroyGrabLine")
    if not createLine or not destroyLine then return end
    pcall(function()
        createLine:FireServer(hrp, CFrame.new(0, 1e9, 0))
        task.wait()
        destroyLine:FireServer(hrp)
    end)
end

local lineLagThread = nil
local lineLagEnabled = false

local function startLineLag()
    if lineLagEnabled then return end
    lineLagEnabled = true
    
    lineLagThread = coroutine.create(function()
        if not GrabEvents then return end
        local createLine = GrabEvents:FindFirstChild("CreateGrabLine")
        if not createLine then return end
        
        while lineLagEnabled do
            local spawnLocation = Workspace:FindFirstChild("SpawnLocation") 
                or Workspace:FindFirstChild("Spawn") 
                or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"))
            
            if spawnLocation then
                local randomX = math.random(-9e9, 9e9)
                local randomZ = math.random(-9e9, 9e9)
                
                local directions = {
                    CFrame.new(randomX, 0, randomZ),
                    CFrame.new(-randomX, 0, -randomZ),
                    CFrame.new(randomX, 0, -randomZ),
                    CFrame.new(-randomX, 0, randomZ),
                }
                
                for _, pos in pairs(directions) do
                    createLine:FireServer(spawnLocation, pos)
                end
            end
            task.wait()
        end
    end)
    coroutine.resume(lineLagThread)
end

local function stopLineLag()
    if not lineLagEnabled then return end
    lineLagEnabled = false
    if lineLagThread then
        coroutine.close(lineLagThread)
        lineLagThread = nil
    end
end

Players.PlayerRemoving:Connect(function(player)
    if isNotificationOn then
        local currentPlayers = #Players:GetPlayers() - 1
        if currentPlayers < 0 then currentPlayers = 0 end
        sendNotification(player.Name, currentPlayers)
    end
end)

local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)

local fps = 0
local lastTick = tick()
local frameCount = 0

RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    local currentTick = tick()
    if currentTick - lastTick >= 1 then
        fps = math.floor(frameCount / (currentTick - lastTick))
        frameCount = 0
        lastTick = currentTick
    end
    
    local timeNow = os.time()
    local jstTime = os.date("!*t", timeNow + (9 * 60 * 60))
    local timeString = string.format("%02d:%02d:%02d", jstTime.hour, jstTime.min, jstTime.sec)
    
    local playerCount = #Players:GetPlayers()
    local maxPlayers = Players.MaxPlayers
    
    textLabel.Text = string.format('<b>narmi hub </b>   |   FPS: %d   |   Time: %s   |   Players: %d/%d', fps, timeString, playerCount, maxPlayers)
end)
