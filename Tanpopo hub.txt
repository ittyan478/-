-- Resonance-style player label: DisplayName + @Name + id
local function FormatPlayer(plr, multiline)
	if typeof(plr) == "string" then
		plr = Players:FindFirstChild(plr)
	end
	if not plr or not plr:IsA("Player") then
		return "???"
	end
	if multiline then
		return string.format("%s\n@%s | %d", plr.DisplayName, plr.Name, plr.UserId)
	end
	return string.format("%s (@%s) [%d]", plr.DisplayName, plr.Name, plr.UserId)
end

local function BuildPlayerValues()
	local values = {}
	local images = {}
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer then
			local label = FormatPlayer(plr)
			table.insert(values, label)
			images[label] = {
				Url = string.format("rbxthumb://type=AvatarHeadShot&id=%s&w=48&h=48", tostring(plr.UserId)),
			}
		end
	end
	table.sort(values)
	return values, images
end

local function LabelToPlayer(label)
	if typeof(label) == "Instance" and label:IsA("Player") then
		return label
	end
	local s = tostring(label or "")
	-- match (@Name) [id] or just Name
	local name = s:match("%(@([^%)]+)%)")
	if name then
		return Players:FindFirstChild(name)
	end
	local id = tonumber(s:match("%[(%d+)%]"))
	if id then
		for _, p in ipairs(Players:GetPlayers()) do
			if p.UserId == id then return p end
		end
	end
	return Players:FindFirstChild(s)
end

function tbl12.Teleport(arg, cFrame2, arg2, arg3)
	-- Resonance exact
	if not arg or not arg:IsA("BasePart") or not cFrame2 then
		return false
	end
	if typeof(cFrame2) == "CFrame" then
		if arg2 then
			cFrame2 = cFrame2 * arg2
		end
	elseif typeof(cFrame2) == "Vector3" then
		if arg2 then
			cFrame2 = cFrame2 + arg2.Position
		end
		cFrame2 = CFrame.new(cFrame2)
	elseif typeof(cFrame2) == "Instance" and cFrame2:IsA("BasePart") then
		if arg2 then
			cFrame2 = cFrame2.CFrame * arg2
		else
			cFrame2 = cFrame2.CFrame
		end
	else
		return false
	end
	if arg.CFrame ~= cFrame2 then
		arg.CFrame = cFrame2
	end
	if arg3 ~= false then
		tbl12.ResetVelocity(arg)
	end
	return true
end

function tbl12.PivotTo(model, cf)
	if not model or not model:IsA("Model") or typeof(cf) ~= "CFrame" then return false end
	local delta = cf * model:GetPivot():Inverse()
	for _, p in ipairs(model:GetDescendants()) do
		if p:IsA("BasePart") and not p.Anchored then
			local root = p.AssemblyRootPart
			if root and not root.Anchored then p.CFrame = delta * p.CFrame end
		end
	end
	return true
end

local velCache = setmetatable({}, { __mode = "k" })
function tbl12.GetCustomVelocity(part)
	local now = os.clock()
	local pos = part.Position
	local e = velCache[part]
	if not e then
		velCache[part] = { Position = pos, Time = now, Linear = Vector3.zero }
		return Vector3.zero
	end
	local dt = now - e.Time
	if dt < 0.001 then return e.Linear end
	local lin = (pos - e.Position) / dt
	if dt > 0.25 or lin.Magnitude > 300 then
		e.Linear = Vector3.zero
	else
		e.Linear = e.Linear:Lerp(lin, 1 - math.exp(-dt * 40))
		if e.Linear.Magnitude < 0.05 then e.Linear = Vector3.zero end
	end
	e.Position, e.Time = pos, now
	return e.Linear
end

function tbl12.SNO(part, timeout)
	local char, root, hum = tbl12.Char()
	local head = char and char:FindFirstChild("Head")
	local rag = hum and hum:FindFirstChild("Ragdolled")
	if not (head and rag and root and hum and hum.Health > 0) then return false end
	if not part or not part:IsA("BasePart") or not part.Parent then return false end
	if tbl12.IsPartOwner(part) then return true end
	if tbl7.States.Owning then return false end

	local far = (root.Position - part.Position).Magnitude >= 31
	tbl7.States.Owning = true
	local savedRoot, savedHead, hold = root.CFrame, head.CFrame, root.CFrame
	local conn
	local function lock()
		savedRoot, savedHead, hold = root.CFrame, head.CFrame, root.CFrame
		RunService:BindToRenderStep("ErisClient", Enum.RenderPriority.Camera.Value - 1, function()
			root.CFrame = savedRoot
			head.CFrame = savedHead
		end)
		conn = RunService.Heartbeat:Connect(function()
			savedRoot, savedHead = root.CFrame, head.CFrame
			root.CFrame, head.CFrame = hold, hold
		end)
	end
	local function unlock()
		pcall(function() RunService:UnbindFromRenderStep("ErisClient") end)
		if conn then conn:Disconnect() end
		head.CFrame, root.CFrame = savedHead, savedRoot
	end

	local ok, err = pcall(function()
		if far then lock() end
		timeout = timeout or 0.65
		local t0 = tick()
		while not flag11 and part.Parent and not tbl12.IsPartOwner(part) and hum.Health ~= 0 and root.Parent and tick() - t0 < timeout do
			if far then hold = tbl12.PredictCFrame(part) - Vector3.new(0, 22, 0) end
			RunService.Heartbeat:Wait()
			tbl12.Grab(part, far and hold or nil)
		end
	end)
	if far then pcall(unlock) end
	tbl7.States.Owning = false
	if not ok then warn(err) return false end
	return tbl12.IsPartOwner(part)
end

			if d:IsA("BasePart") then
				table.insert(baseParts, d)

		if v24.Parent then
			-- Connected StickyPart
			local ok2, connected = pcall(function()
				return v24:GetConnectedParts()
			end)
			if ok2 and connected then
				for _, v25 in pairs(connected) do
					if v25.Name == "StickyPart" or v25.Name == "StickyWeld" then
						for _, child in pairs(v25:GetChildren()) do
							if child:IsA("TouchInterest") or child:IsA("WeldConstraint") or child:IsA("Weld") then
								pcall(function() child:Destroy() end)
							end
						end
						local parent2 = v25.Parent and tbl12.FindNicePart(v25.Parent)
						if parent2 then
							local plotItems = workspace:FindFirstChild("PlotItems")
							if not plotItems or not parent2:IsDescendantOf(plotItems) then
								tbl12.Grab(parent2)
							end
						end
					end
				end
			end
			-- WeldConstraint named StickyWeld on the part
			for _, ch in pairs(v24:GetChildren()) do
				if ch.Name == "StickyWeld" or (ch:IsA("WeldConstraint") and ch.Name:lower():find("sticky")) then
					pcall(function() ch:Destroy() end)
				end
			end
		end
	end

	-- Direct descendant StickyPart on character models
	for _, d in ipairs(arg:GetDescendants()) do
		if d.Name == "StickyPart" then
			for _, child in pairs(d:GetChildren()) do
				if child:IsA("TouchInterest") or child:IsA("WeldConstraint") or child:IsA("Weld") then
					pcall(function() child:Destroy() end)
				end
			end
			local parent2 = d.Parent and tbl12.FindNicePart(d.Parent)
			if parent2 then
				local plotItems = workspace:FindFirstChild("PlotItems")
				if not plotItems or not parent2:IsDescendantOf(plotItems) then

		return
	end
	local humanoidRootPart = arg:FindFirstChild("HumanoidRootPart") or arg:FindFirstChild("Torso") or arg:FindFirstChild("realHrp")

		return
	end

	-- キックで真上に飛ばすと距離が開くので範囲を広げる

	for _, v25 in inv:GetChildren() do
		if v25:IsA("Model") then
			local pivotOk, pivot = pcall(function() return v25:GetPivot() end)
			if not pivotOk or not pivot then continue end

				if v26 then
					tbl12.ToggleCollisions(v25, false)

					tbl12.PivotTo(v25, CFrame.new(0, -50000, 0))
					task.delay(1, tbl12.RNO, v26)
				end
			end
		end
	end

-- Kick start => Nile Line Lag ON
-- Kick専用 Line Lag (Nileと同じ CreateGrabLine) · UIトグルと独立して確実に回す
local kickLineLagToken = 0
local kickLineLagActive = false

local function runNileLineLagBurst()
	-- Nile exact: CreateGrabLine(SpawnLocation, CFrame(-2.3e9,0,-2.3e9))
	local n = tonumber(LagLinesAmount) or 5
	local spawn = Workspace:FindFirstChild("SpawnLocation")
	for _ = 1, n do
		pcall(function()
			CreateGrabLine:FireServer(spawn, CFrame.new(-2.3e9, 0, -2.3e9))
		end)
	end
end

local function startKickLineLag()
	if kickLineLagActive then return end
	kickLineLagActive = true
	kickLineLagToken += 1
	local token = kickLineLagToken
	task.spawn(function()
		while kickLineLagActive and token == kickLineLagToken and not flag11 do
			runNileLineLagBurst()
			task.wait(0.001) -- Nile same interval
		end
	end)
end

local function stopKickLineLag()
	kickLineLagActive = false
	kickLineLagToken += 1
end

local function setKickLag(on)
	if on then
		-- キック中は必ず Nile Line Lag を回す（Lagタブのトグルに依存しない）
		startKickLineLag()
	else
		if not tbl8.LoopApplyMethodT and not tbl8.LoopApplyMethodTServer then
			stopKickLineLag()
		end
	end
	tbl8.RagdollNow = on
end

		local ok = root and hum and hum.Health > 0

		end

	end

		conn = RunService.Heartbeat:Connect(function()
			for part, cf in pairs(locked) do
				if part.Parent then
					part.CFrame = cf
					tbl12.ResetVelocity(part)
				else
					locked[part] = nil
				end
			end
		end)
		for i, p in ipairs(targets) do
			if flag11 then break end
			local _, root, hum = tbl12.Char(p)
			if root and hum and hum.Health > 0 then
				if useCooldown then tbl7.KickServerCooldowns[p] = os.clock() end
				if tbl12.SNO(root) then
					-- 真上キック (縦スタック)
					local cf = CFrame.new(origin + Vector3.new(0, tbl9.NKOffset + (i - 1) * 3, 0))
					table.insert(roots, root)
					locked[root] = cf
					tbl12.Teleport(root, cf)
					pcall(tbl12.NotifyKick, p)
				end
			end
		end
		RunService.Heartbeat:Wait()
	end)
	task.spawn(function()
		local untilT = os.clock() + 3
		while not flag11 and os.clock() < untilT do RunService.Heartbeat:Wait() end
		if conn then conn:Disconnect() end
	end)
	task.wait(0.15)
	for _, root in ipairs(roots) do if root.Parent then tbl12.RNO(root) end end
	for _, root in ipairs(roots) do
		if not flag11 and root.Parent then
			task.spawn(tbl12.RemoveSticky, root.Parent)
		end
	end
	task.wait(6)
	setKickLag(false)
	state.Running = false
	return #roots > 0
end

-- Load preset into ACRotation / AutoClickerOffset (Resonance LoadACEditorPose simplified without UI sliders)
function tbl12.LoadACPreset(pose)
	if not pose then return false end
	local bodyOffset = pose.BodyOffset
	if not bodyOffset and pose.BodyPos then
		bodyOffset = CFrame.new(tonumber(pose.BodyPos[1]) or 0, tonumber(pose.BodyPos[2]) or 0, tonumber(pose.BodyPos[3]) or -5)
	end
	if bodyOffset then
		tbl9.AutoClickerOffset = bodyOffset
	end
	local rotation = pose.Rotation or pose
	for k in pairs(tbl7.ACRotation) do
		local v = rotation[k]
		if v then
			if typeof(v) == "Vector3" then
				tbl7.ACRotation[k] = v
			else
				tbl7.ACRotation[k] = Vector3.new(tonumber(v[1]) or 0, tonumber(v[2]) or 0, tonumber(v[3]) or 0)
			end
		end
	end
	if pose.LimbOffsets then
		for k in pairs(tbl7.ACOffset) do
			local v = pose.LimbOffsets[k]
			if v then
				if typeof(v) == "Vector3" then
					tbl7.ACOffset[k] = v
				else
					tbl7.ACOffset[k] = Vector3.new(tonumber(v[1]) or 0, tonumber(v[2]) or 0, tonumber(v[3]) or 0)
				end
			end
		end
	end
	return true
end

	if p:IsA("BasePart") then

local function scaleNumberSeq(seq, mul)
	if typeof(seq) ~= "NumberSequence" then return seq end

	return NumberSequence.new(kps)

		w.Parent = model
	end

	cacheScale(model)
	for _, d in pairs(model:GetDescendants()) do
		cacheScale(d)
	end

	-- 本家: 回転しながら追従、Hole消滅 or 数秒後に縮小して消える
	local spinSpeed = 40
	local angle = 0
	local shrinking = false
	local life = 1
	local shrinkRate = 1.2
	local maxLife = 5 -- 数秒で消える
	local born = os.clock()
	local hb

	local function beginShrink()
		if shrinking then return end
		shrinking = true
		for _, d in pairs(model:GetDescendants()) do
			if d:IsA("BasePart") then d.Anchored = true end
		end
		if model:IsA("BasePart") then model.Anchored = true end
	end

	hole.AncestryChanged:Connect(function(_, parent)
		if not parent then beginShrink() end
	end)

	hb = RunService.Heartbeat:Connect(function(dt)
		if not model or not model.Parent then
			if hb then hb:Disconnect() end
			return
		end

		-- 数秒経過でも消滅開始
		if not shrinking and os.clock() - born >= maxLife then
			beginShrink()
		end

		if not shrinking and hole and hole.Parent then
			angle = (angle + dt * spinSpeed) % 360
			hole.CFrame = lockedCF * CFrame.Angles(0, math.rad(angle), 0)
			hole.AssemblyLinearVelocity = Vector3.zero
			hole.AssemblyAngularVelocity = Vector3.zero
		end

		if shrinking then
			life = life - dt * shrinkRate
			if life <= 0 then
				if hb then hb:Disconnect() end
				pcall(function() model:Destroy() end)
				return
			end
			for obj, data in pairs(sizeCache) do
				if obj and obj.Parent then
					if obj:IsA("BasePart") and data.Size then
						obj.Size = data.Size * life
					elseif (obj:IsA("SpecialMesh") or obj:IsA("BlockMesh") or obj:IsA("CylinderMesh")) and data.Scale then
						obj.Scale = data.Scale * life
					elseif obj:IsA("Attachment") and data.Position then
						obj.Position = data.Position * life
					elseif obj:IsA("ParticleEmitter") then
						if data.Size then obj.Size = scaleNumberSeq(data.Size, life) end
					elseif obj:IsA("Beam") then
						if data.Width0 then obj.Width0 = data.Width0 * life end
						if data.Width1 then obj.Width1 = data.Width1 * life end
					elseif data.Range then
						obj.Range = data.Range * life
					end
				end
			end
		end
	end)
	table.insert(kickTexHB, hb)
end

local function startHipDanceV2(arg)
	if not FunDance.activeDanceDataV2 then return end
	stopHipDanceV2(arg)
	local root = arg:WaitForChild("HumanoidRootPart")
	local h = arg:WaitForChild("Humanoid")
	if RagdollRemote then
		pcall(function() RagdollRemote:FireServer(root, 9999999) end)
	end
	local position = root.Position
	local ws = h.WalkSpeed > 0 and h.WalkSpeed or 16
	local fc = #FunDance.activeDanceDataV2.frames
	if fc == 0 then return end

	local currentYaw = math.atan2(-root.CFrame.LookVector.X, -root.CFrame.LookVector.Z)
	FunDance.frameIndexV2 = 1
	FunDance.frameTimerV2 = 0
	FunDance.poseConnectionV2 = RunService.Heartbeat:Connect(function(dt)
		if not arg.Parent or not root.Parent then return end
		local md = h.MoveDirection
		if md.Magnitude > 0.01 then
			currentYaw = math.atan2(-md.X, -md.Z)
		end
		if (root.Position - position).Magnitude > 8 then
			position = root.Position
		end
		if md.Magnitude > 0.01 then
			position += md * ws * dt
		end
		local base = CFrame.new(position) * CFrame.Angles(0, currentYaw, 0)
		root.CFrame = base
		root.AssemblyLinearVelocity = Vector3.zero

		FunDance.frameTimerV2 = FunDance.frameTimerV2 + dt
		local wr = false
		if FunDance.activeDanceDataV2.interval <= FunDance.frameTimerV2 then
			FunDance.frameTimerV2 = FunDance.frameTimerV2 - FunDance.activeDanceDataV2.interval
			FunDance.frameIndexV2 = FunDance.frameIndexV2 + 1
			if fc < FunDance.frameIndexV2 then
				FunDance.frameIndexV2 = 1
				wr = true
			end
		end

		local fA = FunDance.activeDanceDataV2.frames[FunDance.frameIndexV2]
		local fB = FunDance.activeDanceDataV2.frames[FunDance.frameIndexV2 < fc and FunDance.frameIndexV2 + 1 or 1]
		local alpha = wr and 1 or (FunDance.frameTimerV2 / FunDance.activeDanceDataV2.interval)

		for pn, cfd in pairs(fA.parts) do
			if not SKIP_PARTS[pn] then
				local p = arg:FindFirstChild(pn)
				if p and p:IsA("BasePart") then
					local od = fB.parts[pn] or cfd
					p.CFrame = base * parseCFrameData(cfd):Lerp(parseCFrameData(od), alpha)
					p.AssemblyLinearVelocity = Vector3.zero
				end
			end
		end
	end)
end

		return game:HttpGet("https://raw.githubusercontent.com/dcus-official/PoopHub/refs/heads/main/Assets/Emote/List_Json.lua")
	end)
	if ok and result then
		local ok2, r2 = pcall(function()

			if ch then return ch() end
		end)
		if ok2 and type(r2) == "table" then
			for k, v in pairs(r2) do
				nt[k] = v
			end
		end
	end
	pcall(function()
		if not isfolder("tanpopoHub") then makefolder("tanpopoHub") end
		if not isfolder("tanpopoHub/FTAP") then makefolder("tanpopoHub/FTAP") end
		if not isfolder("tanpopoHub/FTAP/Emote") then makefolder("tanpopoHub/FTAP/Emote") end
	end)
	local ok2, files = pcall(function()
		return listfiles("tanpopoHub/FTAP/Emote")
	end)
	if ok2 and type(files) == "table" then
		for _, fp in ipairs(files) do
			if fp:match("%.dance$") then
				local name = fp:gsub("\\", "/"):match("([^/]+)%.dance$")
				if name then
					local ok3, c = pcall(readfile, fp)
					if ok3 and c then nt[name] = c end
				end
			end
		end
	end

		if #FunDance.danceListV2 == 0 then
			FunDance.danceListV2 = { "No dances found" }
		end
		return FunDance.danceListV2

	for k in pairs(FunDance.currentDanceRawTableV2) do
		table.insert(nl, k)
	end
	table.sort(nl)
	FunDance.danceListV2 = nl
	return nl

-- Line lag detect (Resonance style)
function tbl12.LineLagHandler(arg, lastPart, arg2)
	if typeof(lastPart) ~= "Instance" or typeof(arg2) ~= "CFrame" or arg == LocalPlayer then
		return
	end
	if not tbl7.NotificationSettings.Events["Line Lag"] then return end
	local tbl16 = tbl7.LineData[arg]
	if not tbl16 then
		tbl16 = {
			Count = 0, FirstTime = 0, LastPart = nil, PartSwitches = 0,
			IsLagging = false, LastTime = 0, Initialized = false, NotificationCooldown = 0,
		}
		tbl7.LineData[arg] = tbl16
	end
	local now3 = tick()
	local v24 = tbl12.FormatPlayer(arg)
	if not tbl16.Initialized then
		tbl16.FirstTime = now3
		tbl16.Initialized = true
		return
	end
	if now3 - tbl16.FirstTime < 1 then return end
	local position = arg2.Position
	local n38 = math.abs(position.X) + math.abs(position.Y) + math.abs(position.Z)
	local flag12 = lastPart and lastPart:IsA("BasePart") and lastPart.CanQuery == false
	if tbl16.LastPart and tbl16.LastPart ~= lastPart then
		tbl16.PartSwitches = tbl16.PartSwitches + 1
	end
	tbl16.LastPart = lastPart
	if now3 - tbl16.FirstTime > 2 then
		tbl16.FirstTime = now3
		tbl16.Count = 0
		tbl16.PartSwitches = 0
	end
	tbl16.Count = tbl16.Count + 1

local function ExpandSticky(tool, cfg, mult)
	if not cfg or not cfg.EXPAND_STICKY then return end
	local sp = tool:FindFirstChild("StickyPart")
	if not sp or expandedStickies[sp] then return end
	expandedStickies[sp] = {
		size = sp.Size,
		transparency = sp.Transparency,
		cantouch = sp.CanTouch,
		cancollide = sp.CanCollide,
		canquery = sp.CanQuery,
	}
	pcall(function() sp.Size = Vector3.new(sp.Size.X, sp.Size.Y * mult, sp.Size.Z) end)
	pcall(function() sp.CanTouch = true end)
	pcall(function() sp.CanCollide = false end)
	pcall(function() sp.CanQuery = true end)
end

local function RestoreSticky(tool)
	local sp = tool and tool:FindFirstChild("StickyPart")
	if not sp then return end
	local data = expandedStickies[sp]
	if not data then return end
	pcall(function() sp.Size = data.size end)
	pcall(function() sp.Transparency = data.transparency end)
	pcall(function() sp.CanTouch = data.cantouch end)
	pcall(function() sp.CanCollide = data.cancollide end)
	pcall(function() sp.CanQuery = data.canquery end)
	expandedStickies[sp] = nil
end

local function StickTool(tool, cfg, toolKey)
	-- pastefy そのまま
	if not tool or not tool:FindFirstChild("StickyPart") then return end
	local currentHRP = AK_getHRP()
	if not currentHRP then return end
	local firePart = currentHRP:FindFirstChild("FirePlayerPart") or currentHRP:WaitForChild("FirePlayerPart", 5)
	if not firePart then return end

	for _, obj in pairs(tool:GetChildren()) do
		if obj:IsA("BasePart") then
			obj.CanTouch = false
			obj.CanCollide = false
			obj.CanQuery = false
			obj.AssemblyLinearVelocity = Vector3.zero
			obj.AssemblyAngularVelocity = Vector3.zero
		end
	end

	if tool:FindFirstChild("SoundPart") and SetNetworkOwner then
		local sp = tool.SoundPart
		if not sp:FindFirstChild("PartOwner") or sp.PartOwner.Value ~= LocalPlayer.Name then
			pcall(function() SetNetworkOwner:FireServer(sp, sp.CFrame) end)
		end
	end

	local mult = currentMult[toolKey] or 0.5
	ExpandSticky(tool, cfg, mult)

	local attachOffset = CFrame.new(cfg.POS_X or 0, cfg.POS_Y or 0, cfg.POS_Z or 0)
	local attachAngle = BuildAttachAngle(cfg)
	local attachCombined = attachOffset * attachAngle
	local targetCFrame = firePart.CFrame * attachCombined
	pcall(function() tool:PivotTo(targetCFrame) end)
	if StickyPartEvent then
		pcall(function()
			StickyPartEvent:FireServer(tool.StickyPart, firePart, attachCombined)
		end)
	end
end

function startAntiKick(toolKey)
	shurikenAntiKickActive = true
	KickCfg.ToolAntiKick = true
	toolKey = toolKey or KickCfg.antiKickToolName or "ToolPickaxe"
	KickCfg.antiKickToolName = toolKey

	if not StickyPartEvent or not SpawnToyRF then
		Library:Notify({ Title = "Anti Kick", Description = "Remote が見つかりません", Time = 3 })
		shurikenAntiKickActive = false
		return
	end

	local cfg = TOOL_CONFIGS[toolKey]
	if not cfg then
		Library:Notify({ Title = "Anti Kick", Description = "ツール設定が見つかりません", Time = 3 })
		shurikenAntiKickActive = false
		return
	end
	local realName = GetRealName(toolKey, cfg)
	currentMult[toolKey] = cfg.STICKY_SIZE_MULT_BASE or 0.5

	task.spawn(function()
		while shurikenAntiKickActive and not flag11 do
			task.wait(0.05)
			local char = LocalPlayer.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not char or not hum or hum.Health <= 0 then continue end

			local inv = Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
			local tool = inv and (inv:FindFirstChild("AntiKick") or inv:FindFirstChild(realName))

			local plotItems = Workspace:FindFirstChild("PlotItems")
			local pip = plotItems and plotItems:FindFirstChild("PlayersInPlots")

			-- 自宅プロット内
			if pip and pip:FindFirstChild(LocalPlayer.Name) then
				local boolik, house = AK_CheckForHome()
				if boolik and house and Workspace.Plots and Workspace.Plots:FindFirstChild(house.Name) then
					if not tool then
						tool = house:FindFirstChild("AntiKick") or house:FindFirstChild(realName)
					end
					if not tool then
						tool = SpawnToyAK(realName)
					end
					if tool then
						tool.Name = "AntiKick"
						-- 溶接チェック後のみ Stick（毎フレ Pivot しない）
						local sp = tool:FindFirstChild("StickyPart")
						if sp then
							local isWelded = sp:FindFirstChild("StickyWeld") and sp.StickyWeld.Part1 ~= nil
							if not isWelded then
								StickTool(tool, cfg, toolKey)
								task.wait(0.15)
							else
								RestoreSticky(tool)
								currentMult[toolKey] = cfg.STICKY_SIZE_MULT_BASE or 0.5
							end
							local hrp = char:FindFirstChild("HumanoidRootPart")
							if hrp and (hrp.Position - sp.Position).Magnitude >= 20 then
								ClearAntiKickTools()
							end
						end
					end
				end
				continue
			end

			-- 通常
			if not tool then
				tool = SpawnToyAK(realName)
				if tool == nil then continue end
				tool.Name = "AntiKick"
			end

			if tool and tool:FindFirstChild("StickyPart") then
				local sp = tool.StickyPart
				local isWelded = sp:FindFirstChild("StickyWeld") and sp.StickyWeld.Part1 ~= nil
				-- 本家: 未溶接のときだけ StickTool（動いて見えるのを防ぐ）
				if not isWelded then
					local step = cfg.STICKY_SIZE_MULT_STEP or 0.5
					local maxM = cfg.STICKY_SIZE_MULT_MAX or 20
					if cfg.EXPAND_STICKY and (currentMult[toolKey] or 0) < maxM then
						currentMult[toolKey] = (currentMult[toolKey] or 0.5) + step
						RestoreSticky(tool)
					end
					StickTool(tool, cfg, toolKey)
					tool.Name = "AntiKick"
					task.wait(0.15)
				else
					RestoreSticky(tool)
					currentMult[toolKey] = cfg.STICKY_SIZE_MULT_BASE or 0.5
				end
				local hrp = char:FindFirstChild("HumanoidRootPart")
				if hrp and (hrp.Position - sp.Position).Magnitude >= 20 then
					ClearAntiKickTools()
				end
			end
		end
		ClearAntiKickTools()
	end)
end

local function GetRainbowColor(offset)
	local t = tick() * 0.5 + (offset or 0)
	local r = math.sin(t) * 0.5 + 0.5
	local g = math.sin(t + 2.094) * 0.5 + 0.5
	local b = math.sin(t + 4.188) * 0.5 + 0.5
	return Color3.new(r, g, b)
end

local function RemoveToolHighlight(model)
	local h = ToolESP.highlights[model]
	if h then
		pcall(function() h:Destroy() end)
		ToolESP.highlights[model] = nil
	end
end

local function AddToolHighlight(model)
	-- ESP は虹色 Highlight のみ（見た目・透明度は一切触らない）
	if not ToolESP.enabled then return end
	if ToolESP.highlights[model] then return end
	if not model or not model.Parent then return end
	if not ToolESP.targets[model.Name] then return end
	local high = Instance.new("Highlight")
	high.Name = "ToolESP_Highlight"
	high.Adornee = model
	high.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	high.FillTransparency = 0.65
	high.OutlineTransparency = 0
	high.FillColor = GetRainbowColor()
	high.OutlineColor = GetRainbowColor(1)
	high.Parent = model
	ToolESP.highlights[model] = high
	model.AncestryChanged:Connect(function(_, parent)
		if not parent then RemoveToolHighlight(model) end
	end)
end

local function ScanFolderForTools(folder)
	if not folder then return end
	for _, v in ipairs(folder:GetChildren()) do
		if ToolESP.targets[v.Name] then
			AddToolHighlight(v)
		end
	end
end

local function SetupFolderConnection(folder)
	if not folder or ToolESP.connections[folder] then return end
	ToolESP.connections[folder] = folder.ChildAdded:Connect(function(child)
		if ToolESP.targets[child.Name] then
			task.defer(AddToolHighlight, child)
		end
	end)
	ScanFolderForTools(folder)
end

local function AI_startAntiInput()
	if AIConfig.loopActive then return end
	AIConfig.loopActive = true
	AI_destroyExistingToy()

	local function ensureToy()
		local invF = Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
		if not invF then return nil end
		local toy = invF:FindFirstChild(AIConfig.targetToy) or invF:FindFirstChild("AntiInputToy")

		local _, root = tbl12.Char()
		if not root then return nil end

		pcall(function()
			SpawnToyRF:InvokeServer(AIConfig.targetToy, root.CFrame * CFrame.new(0, 14, 20), Vector3.zero)
		end)
		toy = invF:WaitForChild(AIConfig.targetToy, 2.5)
		if toy then toy.Name = "AntiInputToy" end
		return toy
	end

	AIConfig.runLoop = RunService.Heartbeat:Connect(function()
		if not AIConfig.loopActive or flag11 then return end
		local toy = ensureToy()
		if not toy then return end
		local char = LocalPlayer.Character
		local head = char and char:FindFirstChild("Head")
		if not head then return end
		local holdPart = toy:FindFirstChild("HoldPart")

		local dropRemote = holdPart:FindFirstChild("DropItemRemoteFunction")

-- ===== 8-Bit Font（サーバー全体の数字のみ · タブアイコンは触らない） =====
tbl8.EightBitFont = true

local function isMostlyDigits(text)
	if type(text) ~= "string" or text == "" then return false end
	local stripped = text:gsub("[%d%.,:%-%+%%%$%s/]", "")
	if #stripped > 0 then return false end
	return text:find("%d") ~= nil
end

local function applyNumberFont(obj)
	if not obj or not obj.Parent then return end
	if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then return end
	-- スクリプト UI（Obsidian / タブ）は完全除外
	local p = obj
	for _ = 1, 18 do
		if not p then break end
		local n = (p.Name or ""):lower()
		if n == "obsidian" or n == "library" or n == "main" or n == "sidebar"
			or n:find("sidebar") or n:find("tab") or n:find("icon")
			or n:find("lucide") or n == "notifs" or n == "notifications" then
			return
		end
		-- ScreenGui が Library のもの
		if p:IsA("ScreenGui") and Library and Library.ScreenGui and p == Library.ScreenGui then
			return
		end
		p = p.Parent
	end
	local t = ""
	pcall(function() t = obj.Text end)
	if isMostlyDigits(t) then
		pcall(function()
			obj.Font = Enum.Font.Arcade
		end)
	end
end

function tbl12.Apply8BitScriptUI()
	-- タブアイコン保護のためスクリプトUIには当てない
end

function tbl12.Apply8BitNumbers(root)
	if not root then return end
	for _, d in ipairs(root:GetDescendants()) do
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			applyNumberFont(d)
		end
	end
end

function tbl12.ExecuteKill(t)
	-- pastefy ExecuteKill そのまま
	if not t or not t.Char or not t.Char.Parent then return end

	if not getgenv().LK_OriginalFallenHeight then
		getgenv().LK_OriginalFallenHeight = Workspace.FallenPartsDestroyHeight
	end
	Workspace.FallenPartsDestroyHeight = 0 / 0

	local returnCF = tbl7.LoopKill.ReturnCFrame
	local myHrp = LK_HRP()
	if returnCF and myHrp then
		myHrp.CFrame = returnCF
		myHrp.AssemblyLinearVelocity = Vector3.zero
		myHrp.AssemblyAngularVelocity = Vector3.zero
	end

	tbl7.LoopKill.KillLock = true

	pcall(function()
		for _, p in pairs(t.Char:GetDescendants()) do
			if p:IsA("BasePart") then
				p.CanCollide = false
				p.Massless = true
				p.CanTouch = false
				p.CanQuery = false
			end
		end
	end)

	pcall(function()
		for _, d in pairs(t.Char:GetDescendants()) do
			if d:IsA("BodyVelocity") or d:IsA("BodyPosition") or d:IsA("BodyGyro")
				or d:IsA("LinearVelocity") or d:IsA("AlignPosition")
				or d:IsA("AlignOrientation") or d:IsA("VectorForce")
				or d:IsA("BodyThrust") or d:IsA("BodyForce") then
				d.Enabled = false
			end
		end
	end)

	pcall(function()
		t.HRP.Anchored = false
		t.HRP.AssemblyLinearVelocity = Vector3.zero
		t.HRP.AssemblyAngularVelocity = Vector3.zero
	end)

	task.wait()

	pcall(function()
		t.HRP.Anchored = true
		t.HRP.CFrame = CFrame.new(t.HRP.Position.X, -9e15, t.HRP.Position.Z)
	end)

	task.wait()

	pcall(function()
		t.HRP.Anchored = false
		t.HRP.AssemblyLinearVelocity = Vector3.new(0, -9e15, 0)
		t.HRP.AssemblyAngularVelocity = Vector3.new(
			math.random(-1e5, 1e5),
			math.random(-1e5, 1e5),
			math.random(-1e5, 1e5)
		)
	end)

	for _ = 1, 10 do
		task.delay(0, function()
			pcall(function() DestroyGrabLine:FireServer(t.HRP) end)
		end)
	end

	pcall(tbl12.NotifyKick, t.Player)

	task.spawn(function()
		for _ = 1, 10 do
			local h = LK_HRP()
			if returnCF and h then
				h.CFrame = returnCF
				h.AssemblyLinearVelocity = Vector3.zero
				h.AssemblyAngularVelocity = Vector3.zero
			end
			task.wait()
		end
		if getgenv().LK_OriginalFallenHeight then
			Workspace.FallenPartsDestroyHeight = getgenv().LK_OriginalFallenHeight
			getgenv().LK_OriginalFallenHeight = nil
		end
		tbl7.LoopKill.ReturnCFrame = nil
		tbl7.LoopKill.KillLock = false
	end)
end

function tbl12.SingleOwnershipTick()
	-- pastefy · NETWORK_PER_FRAME=100 / NETWORK_DISTANCE=80
	local lk = tbl7.LoopKill
	if not lk.Enabled then return end
	if not lk.OwnershipHeartbeat then return end
	if lk.KillLock then return end

	local targets = tbl12.GetTargets(false)
	if #targets == 0 then return end
	local targetPlr = targets[1]
	if not targetPlr or not targetPlr.Parent then return end

	local char = targetPlr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not char or not hum or hum.Health <= 0 then return end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local head = char:FindFirstChild("Head")
	if not (hrp and head) then return end

	if not lk.ReturnCFrame then
		local myHrp = LK_HRP()
		if myHrp then lk.ReturnCFrame = myHrp.CFrame end
	end

	local mhrp = LK_HRP()
	if not mhrp then return end

	local dir = (hrp.Position - mhrp.Position)
	if dir.Magnitude < 0.01 then dir = Vector3.new(0, 0, -1) else dir = dir.Unit end
	local right = dir:Cross(Vector3.new(0, 1, 0))
	if right.Magnitude < 0.01 then right = Vector3.new(1, 0, 0) else right = right.Unit end
	local up = right:Cross(dir)
	local netCFrame = CFrame.fromMatrix(mhrp.Position, right, up)
	local distance = LocalPlayer:DistanceFromCharacter(hrp.Position)

	-- 掴む量: 1フレームに 100 回 FireServer（本家そのまま）
	if distance <= NETWORK_DISTANCE then
		for _ = 1, NETWORK_PER_FRAME do
			pcall(function()
				SetNetworkOwner:FireServer(hrp, netCFrame)
			end)
		end
	end

	pcall(function()
		local truePos = char:FindFirstChild("TruePositionPart")
		if truePos then truePos:Destroy() end
		local attach = hrp:FindFirstChild("RootAttachment")
		if attach then attach.Parent = hrp end
	end)

	pcall(function()
		DestroyGrabLine:FireServer(hrp)
	end)

	pcall(function()
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
		hrp.Anchored = false
	end)

	pcall(function()
		local partOwner = head:FindFirstChild("PartOwner")
		if partOwner then partOwner.Value = LocalPlayer.Name end
	end)

	local hasOwner = false
	if head:FindFirstChild("PartOwner") and head.PartOwner.Value == LocalPlayer.Name then
		hasOwner = true
	end

	if hasOwner or hrp.AssemblyLinearVelocity.Magnitude > 500 then
		if not lk.OwnershipHeartbeat then return end
		LK_StopOwnership()
		lk.StartTime = 0
		tbl12.ExecuteKill({
			Player = targetPlr,
			Char = char,
			HRP = hrp,
			Head = head,
			Hum = hum,
		})
		return
	end

	if LK_HRP() then
		local curMhrp = LK_HRP()
		if hrp.Position.Y <= -12 then
			if lk.ReturnCFrame then
				curMhrp.CFrame = lk.ReturnCFrame
			end
		else
			-- 精度: ターゲット下後方に位置取り
			curMhrp.CFrame = CFrame.new(hrp.Position + Vector3.new(0, -10, -10))
		end
	end

	pcall(function()
		hum.BreakJointsOnDeath = false
		hum.Jump = true
		hum.Sit = false
		hum:ChangeState(Enum.HumanoidStateType.Dead)
	end)
end

local function LK_StartOwnership()
	LK_StopOwnership()
	tbl7.LoopKill.OwnershipHeartbeat = RunService.Heartbeat:Connect(function()
		tbl12.SingleOwnershipTick()
	end)
end

local function LK_TryResume(targetPlr)
	local lk = tbl7.LoopKill
	if not lk.AutoToggleOn then return end
	if not targetPlr then return end
	local char = targetPlr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) or hum.Health <= 0 then return end
	if lk.OwnershipHeartbeat and lk.Enabled then return end
	lk.ReturnCFrame = nil
	lk.KillLock = false
	lk.Enabled = true
	lk.StartTime = tick()
	LK_StartOwnership()
end

function tbl12.StartLoopKill()
	local lk = tbl7.LoopKill
	lk.AutoToggleOn = true
	lk.Enabled = true
	lk.StartTime = tick()
	lk.KillLock = false
	lk.ReturnCFrame = nil
	local targets = tbl12.GetTargets(false)
	if #targets > 0 then
		lk.CurrentTarget = targets[1]
	end
	if lk.AutoToggleConn then
		lk.AutoToggleConn:Disconnect()
		lk.AutoToggleConn = nil
	end
	lk.AutoTogglePhase = false
	lk.AutoToggleConn = RunService.Heartbeat:Connect(function()
		if not lk.AutoToggleOn then return end
		lk.AutoTogglePhase = not lk.AutoTogglePhase
		if lk.AutoTogglePhase then
			local list = tbl12.GetTargets(false)
			if #list == 0 then return end
			local targetPlr = list[1]
			lk.Enabled = true
			lk.ReturnCFrame = nil
			lk.CurrentTarget = targetPlr
			if not lk.OwnershipHeartbeat then
				lk.StartTime = tick()
				LK_StartOwnership()
			end
			-- リスポーン監視
			if targetPlr.Character ~= lk.LastObservedChar then
				lk.LastObservedChar = targetPlr.Character
				local hum = targetPlr.Character and targetPlr.Character:FindFirstChildOfClass("Humanoid")
				if hum and hum.Health > 0 then
					LK_TryResume(targetPlr)
				end
			end
		end
	end)
	LK_StartOwnership()
	Library:Notify({ Title = "Loop Kill", Description = "開始 (100 grab/frame · 80 studs)", Time = 2 })
end

function tbl12.StopLoopKill()
	local lk = tbl7.LoopKill
	lk.AutoToggleOn = false
	lk.Enabled = false
	LK_StopOwnership()
	if lk.AutoToggleConn then
		lk.AutoToggleConn:Disconnect()
		lk.AutoToggleConn = nil
	end
	lk.StartTime = 0
	lk.KillLock = false
	-- 掴み解除
	for _, p in ipairs(tbl12.GetTargets(false)) do
		local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		if hrp then
			task.delay(0.05, function()
				pcall(function() DestroyGrabLine:FireServer(hrp) end)
			end)
		end
	end
	Library:Notify({ Title = "Loop Kill", Description = "停止", Time = 2 })
end

function tbl12.Grab(part, cf)
	if tbl7.LoopKill and tbl7.LoopKill.Enabled and part then
		local mhrp = LK_HRP()
		if mhrp and part:IsA("BasePart") then
			local dir = (part.Position - mhrp.Position)
			if dir.Magnitude < 0.01 then dir = Vector3.new(0, 0, -1) else dir = dir.Unit end
			local right = dir:Cross(Vector3.new(0, 1, 0))
			if right.Magnitude < 0.01 then right = Vector3.new(1, 0, 0) else right = right.Unit end
			local up = right:Cross(dir)
			local netCFrame = CFrame.fromMatrix(mhrp.Position, right, up)
			for _ = 1, NETWORK_PER_FRAME do
				pcall(function()
					SetNetworkOwner:FireServer(part, netCFrame)
				end)
			end
			return
		end
	end
	return _oldGrab(part, cf)
end

function tbl12.FormatPlayer(arg, arg2, arg3)
	local plr
	if type(arg) ~= "string" then
		plr = arg
	else
		plr = Players:FindFirstChild(arg)
	end
	if plr == nil then
		return "???"
	end
	if plr == LocalPlayer and arg3 then
		return "You"
	end
	if arg2 then
		return string.format("%s\n(@%s)", plr.DisplayName, plr.Name)
	end
	return string.format("%s (@%s)", plr.DisplayName, plr.Name)
end

local function makePlayerDropdown(groupbox, name, selectedTable)
	local dd = groupbox:AddDropdown(name, {
		SpecialType = "Player",
		Text = "Targets",
		Default = nil,
		Multi = true,
		AllowNull = true,
		Searchable = true,
		ExcludeLocalPlayer = true,
		EnablePlayerImages = true,
		FormatListValue = function(v)
			return tbl12.FormatPlayer(v)
		end,
		FormatDisplayValue = function(v)
			return tbl12.FormatPlayer(v)
		end,
		Callback = function(val)
			table.clear(selectedTable)
			if type(val) == "table" then
				for k, on in pairs(val) do
					if on then
						local plr = typeof(k) == "Instance" and k or Players:FindFirstChild(tostring(k))
						if plr and plr ~= LocalPlayer then
							selectedTable[plr] = true
						end
					end
				end
			elseif typeof(val) == "Instance" then
				selectedTable[val] = true
			end
		end,
	})
	-- fallback: if SpecialType Player not supported, use Values + ValueImages
	task.defer(function()
		if not dd then return end
		local function refresh()
			local values, images = {}, {}
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= LocalPlayer then
					local label = tbl12.FormatPlayer(p)
					table.insert(values, label)
					images[label] = {
						Url = string.format("rbxthumb://type=AvatarHeadShot&id=%s&w=48&h=48", tostring(p.UserId)),
					}
					-- also map player instance keys if needed
				end
			end
			-- only SetValues if SpecialType didn't populate
			pcall(function()
				if dd.SetValueImages then
					dd:SetValueImages(images)
				end
			end)
		end
		refresh()
		Players.PlayerAdded:Connect(function() task.wait(0.3) refresh() end)
		Players.PlayerRemoving:Connect(function() task.wait(0.3) refresh() end)
	end)
	return dd
end
